// Minimal views of the Stripe objects the handlers read. They are declared here instead of
// using the SDK's types so a Stripe API-version change that moves a field cannot break decoding
// of the fields we do not use. Newer API versions move the subscription link on an invoice under
// parent.subscription_details; both locations are read.

type checkoutSession struct {
        ID                string            `json:"id"`
        Mode              string            `json:"mode"`
        PaymentStatus     string            `json:"payment_status"`
        Customer          string            `json:"customer"`
        Subscription      string            `json:"subscription"`
        PaymentIntent     string            `json:"payment_intent"`
        ClientReferenceID string            `json:"client_reference_id"`
        Metadata          map[string]string `json:"metadata"`
}

type subscriptionDetails struct {
        Subscription string            `json:"subscription"`
        Metadata     map[string]string `json:"metadata"`
}

type invoice struct {
        ID                  string               `json:"id"`
        Subscription        string               `json:"subscription"`
        PaymentIntent       string               `json:"payment_intent"`
        Charge              string               `json:"charge"`
        SubscriptionDetails *subscriptionDetails `json:"subscription_details"`
        Parent              *struct {
                SubscriptionDetails *subscriptionDetails `json:"subscription_details"`
        } `json:"parent"`
        Lines struct {
                Data []struct {
                        Quantity int64 `json:"quantity"`
                        Period   struct {
                                End int64 `json:"end"`
                        } `json:"period"`
                } `json:"data"`
        } `json:"lines"`
}

// details returns the subscription link and metadata from whichever location carries it.
func (i invoice) details() (subscription string, metadata map[string]string) {
        candidates := []*subscriptionDetails{i.SubscriptionDetails}
        if i.Parent != nil {
                candidates = append(candidates, i.Parent.SubscriptionDetails)
        }
        subscription = i.Subscription
        for _, d := range candidates {
                if d == nil {
                        continue
                }
                if subscription == "" {
                        subscription = d.Subscription
                }
                if metadata == nil {
                        metadata = d.Metadata
                }
        }
        return subscription, metadata
}

// quantity is the first line's quantity, 1 if absent.
func (i invoice) quantity() int64 {
        if len(i.Lines.Data) > 0 && i.Lines.Data[0].Quantity > 0 {
                return i.Lines.Data[0].Quantity
        }
        return 1
}

// periodEnd is the latest period end across the invoice's lines, 0 if none.
func (i invoice) periodEnd() int64 {
        var end int64
        for _, l := range i.Lines.Data {
                end = max(end, l.Period.End)
        }
        return end
}

type subscription struct {
        ID       string            `json:"id"`
        Status   string            `json:"status"`
        Metadata map[string]string `json:"metadata"`
        Items    struct {
                Data []struct {
                        Quantity int64 `json:"quantity"`
                        Price    struct {
                                ID string `json:"id"`
                        } `json:"price"`
                } `json:"data"`
        } `json:"items"`
}

// priceID is the first line item's Stripe price ID, "" if there is none.
func (s subscription) priceID() string {
        if len(s.Items.Data) == 0 {
                return ""
        }
        return s.Items.Data[0].Price.ID
}

// quantity is the first line item's quantity, 0 if there is none.
func (s subscription) quantity() int64 {
        if len(s.Items.Data) == 0 {
                return 0
        }
        return s.Items.Data[0].Quantity
}

// sessions going out, signed webhook events coming in as grants.
package billing

import (
        "context"
        "fmt"
        "strconv"
        "time"

        "github.com/stripe/stripe-go/v87"
)

// CheckoutParams is everything needed to start a Checkout session for one purchasable tier.
type CheckoutParams struct {
        PriceID    string
        UserID     string
        TierID     string
        ProductID  string
        Quantity   int64  // units of the price; 0 means 1
        Email      string // optional; Checkout asks for one when empty
        CustomerID string // optional; reuses the user's existing Stripe customer
        SuccessURL string
        CancelURL  string
}

// Gateway is the slice of Stripe's API this service calls. Tests use a fake; StripeGateway is
// the only implementation that touches the network.
type Gateway interface {
        CreateCheckout(ctx context.Context, p CheckoutParams) (url string, err error)
        CreatePortal(ctx context.Context, customerID, returnURL string) (url string, err error)
        // Price reads a price's amount, currency, and interval.
        Price(ctx context.Context, priceID string) (PriceInfo, error)
        // Subscription reads a subscription's line item, quantity, and period end.
        Subscription(ctx context.Context, subscriptionID string) (SubscriptionInfo, error)
        // Invoice reads why an invoice was created (billing_reason).
        Invoice(ctx context.Context, invoiceID string) (InvoiceInfo, error)
        // CancelSubscription ends a subscription immediately: no further invoices, nothing refunded.
        CancelSubscription(ctx context.Context, subscriptionID string) error
        // PreviewSwitch asks Stripe what the switch in p would invoice, without changing anything.
        PreviewSwitch(ctx context.Context, p SwitchParams) (Preview, error)
        // SwitchPlan moves an existing subscription to another price instead of creating a second
        // subscription, with the proration and payment behavior in p.
        SwitchPlan(ctx context.Context, p SwitchParams) error
}

// PriceInfo is what a switch needs to know about a Stripe price.
type PriceInfo struct {
        ID            string
        UnitAmount    int64 // minor units (cents)
        Currency      string
        Interval      string // day, week, month, or year; empty for a one-off price
        IntervalCount int64
}

// SubscriptionInfo is a subscription's single line item as Stripe holds it.
type SubscriptionInfo struct {
        ID               string
        CustomerID       string
        ItemID           string
        Price            PriceInfo
        Quantity         int64
        CurrentPeriodEnd time.Time
}

// Proration behaviors for a plan switch.
const (
        // ProrateCredit credits the unused part of the old plan to the next invoice.
        ProrateCredit = "create_prorations"
        // ProrateInvoiceNow invoices the proration immediately, so an upgrade is paid for now.
        ProrateInvoiceNow = "always_invoice"
)

// InvoiceInfo is what a refund decision needs to know about an invoice.
type InvoiceInfo struct {
        ID string
        // BillingReason is subscription_create or subscription_cycle for the plan's own payments, and
        // subscription_update for a charge made by changing plan.
        BillingReason string
}

// Preview is Stripe's invoice preview for a proposed switch.
type Preview struct {
        Currency string
        Lines    []PreviewLine
}

// PreviewLine is one line of a preview. Proration marks the lines created by the plan change
// itself (the unused time on the old plan and the remaining time on the new), as opposed to the
// renewal that would follow.
type PreviewLine struct {
        Description string
        Amount      int64 // minor units; negative is a credit
        Proration   bool
}

// SwitchParams is one subscription plan change.
type SwitchParams struct {
        SubscriptionID string
        CustomerID     string
        ItemID         string
        Quantity       int64
        PriceID        string
        ProductID      string
        UserID         string
        // Proration is ProrateCredit or ProrateInvoiceNow.
        Proration string
        // ChargeNow makes a failed payment fail the whole update (Stripe's error_if_incomplete)
        // rather than leaving the subscription changed and unpaid.
        ChargeNow bool
        // ProrationDate pins the moment proration is computed (unix seconds); 0 means now.
        ProrationDate int64
}

// StripeGateway calls the live Stripe API. It is a thin translation layer with no rules of its
// own, so it is not unit-tested; the rules live in Service.
type StripeGateway struct{ client *stripe.Client }

// NewStripeGateway builds a gateway authenticated with the secret key.
func NewStripeGateway(secretKey string) *StripeGateway {
        return &StripeGateway{client: stripe.NewClient(secretKey)}
}

// CreateCheckout opens a subscription-mode session for a recurring price and a payment-mode
// session for a one-off price. The user and tier ride along as metadata so webhook events can
// be turned into grants without a lookup.
func (g *StripeGateway) CreateCheckout(ctx context.Context, p CheckoutParams) (string, error) {
        price, err := g.client.V1Prices.Retrieve(ctx, p.PriceID, nil)
        if err != nil {
                return "", fmt.Errorf("retrieve price: %w", err)
        }
        qty := max(p.Quantity, 1)
        meta := map[string]string{"user_id": p.UserID}
        if p.ProductID != "" {
                meta["product_id"] = p.ProductID
                meta["quantity"] = strconv.FormatInt(qty, 10)
        } else {
                meta["tier_id"] = p.TierID
        }
        params := &stripe.CheckoutSessionCreateParams{
                SuccessURL:        stripe.String(p.SuccessURL),
                CancelURL:         stripe.String(p.CancelURL),
                ClientReferenceID: stripe.String(p.UserID),
                Metadata:          meta,
                LineItems:         []*stripe.CheckoutSessionCreateLineItemParams{{Price: stripe.String(p.PriceID), Quantity: stripe.Int64(qty)}},
        }
        if price.Type == stripe.PriceTypeRecurring {
                params.Mode = stripe.String(string(stripe.CheckoutSessionModeSubscription))
                params.SubscriptionData = &stripe.CheckoutSessionCreateSubscriptionDataParams{Metadata: meta}
        } else {
                params.Mode = stripe.String(string(stripe.CheckoutSessionModePayment))
        }
        if p.CustomerID != "" {
                params.Customer = stripe.String(p.CustomerID)
        } else if p.Email != "" {
                params.CustomerEmail = stripe.String(p.Email)
        }
        sess, err := g.client.V1CheckoutSessions.Create(ctx, params)
        if err != nil {
                return "", fmt.Errorf("create checkout session: %w", err)
        }
        return sess.URL, nil
}

// Price reads a price.
func (g *StripeGateway) Price(ctx context.Context, priceID string) (PriceInfo, error) {
        price, err := g.client.V1Prices.Retrieve(ctx, priceID, nil)
        if err != nil {
                return PriceInfo{}, fmt.Errorf("retrieve price: %w", err)
        }
        return priceInfo(price), nil
}

func priceInfo(p *stripe.Price) PriceInfo {
        info := PriceInfo{ID: p.ID, UnitAmount: p.UnitAmount, Currency: string(p.Currency)}
        if p.Recurring != nil {
                info.Interval, info.IntervalCount = string(p.Recurring.Interval), p.Recurring.IntervalCount
        }
        return info
}

// Subscription reads a subscription and its first line item.
func (g *StripeGateway) Subscription(ctx context.Context, subscriptionID string) (SubscriptionInfo, error) {
        sub, err := g.client.V1Subscriptions.Retrieve(ctx, subscriptionID, nil)
        if err != nil {
                return SubscriptionInfo{}, fmt.Errorf("retrieve subscription: %w", err)
        }
        if sub.Items == nil || len(sub.Items.Data) == 0 || sub.Items.Data[0].Price == nil {
                return SubscriptionInfo{}, fmt.Errorf("subscription %s has no priced item", subscriptionID)
        }
        item := sub.Items.Data[0]
        info := SubscriptionInfo{
                ID: sub.ID, ItemID: item.ID, Price: priceInfo(item.Price), Quantity: max(item.Quantity, 1),
                CurrentPeriodEnd: time.Unix(item.CurrentPeriodEnd, 0).UTC(),
        }
        if sub.Customer != nil {
                info.CustomerID = sub.Customer.ID
        }
        return info, nil
}

// Invoice reads an invoice's billing reason.
func (g *StripeGateway) Invoice(ctx context.Context, invoiceID string) (InvoiceInfo, error) {
        inv, err := g.client.V1Invoices.Retrieve(ctx, invoiceID, nil)
        if err != nil {
                return InvoiceInfo{}, fmt.Errorf("retrieve invoice: %w", err)
        }
        return InvoiceInfo{ID: inv.ID, BillingReason: string(inv.BillingReason)}, nil
}

// CancelSubscription cancels now without a final invoice or proration refund; the refund that
// prompted it is left exactly as it is.
func (g *StripeGateway) CancelSubscription(ctx context.Context, subscriptionID string) error {
        _, err := g.client.V1Subscriptions.Cancel(ctx, subscriptionID, &stripe.SubscriptionCancelParams{
                InvoiceNow: stripe.Bool(false), Prorate: stripe.Bool(false),
        })
        if err != nil {
                return fmt.Errorf("cancel subscription: %w", err)
        }
        return nil
}

// PreviewSwitch previews the invoice the switch would produce with the same item change,
// proration behavior, and proration date the real switch would use.
func (g *StripeGateway) PreviewSwitch(ctx context.Context, p SwitchParams) (Preview, error) {
        details := &stripe.InvoiceCreatePreviewSubscriptionDetailsParams{
                Items: []*stripe.InvoiceCreatePreviewSubscriptionDetailsItemParams{{
                        ID: stripe.String(p.ItemID), Price: stripe.String(p.PriceID), Quantity: stripe.Int64(max(p.Quantity, 1)),
                }},
                ProrationBehavior: stripe.String(p.Proration),
        }
        if p.ProrationDate != 0 {
                details.ProrationDate = stripe.Int64(p.ProrationDate)
        }
        inv, err := g.client.V1Invoices.CreatePreview(ctx, &stripe.InvoiceCreatePreviewParams{
                Customer:            stripe.String(p.CustomerID),
                Subscription:        stripe.String(p.SubscriptionID),
                SubscriptionDetails: details,
        })
        if err != nil {
                return Preview{}, fmt.Errorf("preview subscription change: %w", err)
        }
        out := Preview{Currency: string(inv.Currency)}
        if inv.Lines != nil {
                for _, l := range inv.Lines.Data {
                        pl := PreviewLine{Description: l.Description, Amount: l.Amount}
                        if l.Parent != nil && l.Parent.SubscriptionItemDetails != nil {
                                pl.Proration = l.Parent.SubscriptionItemDetails.Proration
                        }
                        out.Lines = append(out.Lines, pl)
                }
        }
        return out, nil
}

// SwitchPlan replaces the subscription item's price, keeping its quantity, and records the new
// product in the subscription's metadata. A charge-now switch fails as a whole if the payment
// does not go through, so Stripe never holds a changed, unpaid subscription.
func (g *StripeGateway) SwitchPlan(ctx context.Context, p SwitchParams) error {
        params := &stripe.SubscriptionUpdateParams{
                Items: []*stripe.SubscriptionUpdateItemParams{{
                        ID: stripe.String(p.ItemID), Price: stripe.String(p.PriceID), Quantity: stripe.Int64(max(p.Quantity, 1)),
                }},
                ProrationBehavior: stripe.String(p.Proration),
                Metadata:          map[string]string{"user_id": p.UserID, "product_id": p.ProductID},
        }
        if p.ChargeNow {
                params.PaymentBehavior = stripe.String("error_if_incomplete")
        }
        if p.ProrationDate != 0 {
                params.ProrationDate = stripe.Int64(p.ProrationDate)
        }
        if _, err := g.client.V1Subscriptions.Update(ctx, p.SubscriptionID, params); err != nil {
                return fmt.Errorf("switch subscription: %w", err)
        }
        return nil
}

// CreatePortal opens the Stripe billing portal for a customer.
func (g *StripeGateway) CreatePortal(ctx context.Context, customerID, returnURL string) (string, error) {
        sess, err := g.client.V1BillingPortalSessions.Create(ctx, &stripe.BillingPortalSessionCreateParams{
                Customer:  stripe.String(customerID),
                ReturnURL: stripe.String(returnURL),
        })
        if err != nil {
                return "", fmt.Errorf("create portal session: %w", err)
        }
        return sess.URL, nil
}

// ProviderError is a failed call to the payment provider (Stripe): an API error, or a transport
// failure such as a timeout. It carries only the fields safe to log and none of Stripe's payload,
// which can hold payment objects. Handlers map it to a 502 with PublicMessage; the cause stays in
// the log.
type ProviderError struct {
        Op        string // checkout, portal, or plan switch
        Type      string // Stripe error type, empty for a transport failure
        Code      string // Stripe error code, e.g. resource_missing
        Param     string
        Message   string // Stripe's message, for the log only
        RequestID string
        Status    int // HTTP status Stripe answered with, 0 for a transport failure
        API       bool
        cause     error
}

func newProviderError(op string, err error) *ProviderError {
        pe := &ProviderError{Op: op, cause: err}
        var se *stripe.Error
        if errors.As(err, &se) {
                pe.API = true
                pe.Type, pe.Code, pe.Param = string(se.Type), string(se.Code), se.Param
                pe.Message, pe.RequestID, pe.Status = se.Msg, se.RequestID, se.HTTPStatusCode
        }
        return pe
}

func (e *ProviderError) Error() string {
        detail := "unreachable"
        if e.API {
                detail = e.Type + " " + e.Code
        }
        return "payment provider " + e.Op + " failed: " + strings.TrimSpace(detail)
}

// Unwrap exposes the underlying error to errors.Is and errors.As.
func (e *ProviderError) Unwrap() error { return e.cause }

// PublicMessage is short text safe to show a caller. A missing price is called out because it is
// a catalog mistake an administrator can fix; everything else stays generic.
func (e *ProviderError) PublicMessage() string {
        switch {
        case !e.API:
                return "the payment provider could not be reached"
        case e.Code == string(stripe.ErrorCodeResourceMissing) && strings.Contains(strings.ToLower(e.Message), "price"):
                return "the product's price is not valid"
        }
        return "the payment provider rejected the request"
}

// providerFailed logs why a provider call failed, with the caller's identifiers, and returns the
// typed error. Only chosen fields are logged: never the key, the secret, or Stripe's full error.
func providerFailed(op string, err error, fields ...any) error {
        pe := newProviderError(op, err)
        args := append([]any{"op", op}, fields...)
        args = append(args, "stripe_type", pe.Type, "stripe_code", pe.Code, "stripe_param", pe.Param,
                "stripe_status", pe.Status, "stripe_request_id", pe.RequestID, "stripe_message", pe.Message)
        if !pe.API {
                args = append(args, "cause", err.Error())
        }
        logging.Logger.Error("payment provider call failed", args...)
        return pe
}

// paymentFailureCodes are Stripe error codes that mean the customer's payment did not go through
// (as opposed to a mistake in the request): a decline, or a card that needs authentication, which
// an off-session charge cannot get.
var paymentFailureCodes = map[string]bool{
        "card_declined":                               true,
        "expired_card":                                true,
        "incorrect_cvc":                               true,
        "insufficient_funds":                          true,
        "authentication_required":                     true,
        "payment_intent_authentication_failure":       true,
        "subscription_payment_intent_requires_action": true,
        "payment_method_provider_decline":             true,
        "processing_error":                            true,
}

// PaymentFailed reports whether the provider refused the charge itself.
func (e *ProviderError) PaymentFailed() bool {
        return e.API && (e.Type == string(stripe.ErrorTypeCard) || paymentFailureCodes[e.Code])
}

// PaymentFailedError means a plan change that had to be paid for now was not paid, and so was not
// made: the subscription and the grants are unchanged.
type PaymentFailedError struct{ cause *ProviderError }

func (e *PaymentFailedError) Error() string { return "payment failed; the plan was not changed" }

// Unwrap exposes the provider error to errors.As.
func (e *PaymentFailedError) Unwrap() error { return e.cause }

// PublicMessage is safe to show a caller.
func (e *PaymentFailedError) PublicMessage() string {
        return "the payment could not be completed, so the plan was not changed"
}

// alreadyCanceled reports whether a cancel failed only because the subscription is already gone
// or already canceled, which is the state the cancel was after.
func alreadyCanceled(err error) bool {
        var se *stripe.Error
        if !errors.As(err, &se) {
                return false
        }
        return se.Code == stripe.ErrorCodeResourceMissing ||
                (se.Type == stripe.ErrorTypeInvalidRequest && strings.Contains(strings.ToLower(se.Msg), "canceled subscription"))
}

// Quote actions.
const (
        QuoteCheckout = "checkout" // a new purchase through Stripe Checkout
        QuoteSwitch   = "switch"   // the user's subscription in the plan group moves to the product
        QuoteNone     = "none"     // the user already holds the product; nothing to buy
)

// QuoteProduct names a product in a quote.
type QuoteProduct struct {
        ID   string `json:"id"`
        Name string `json:"name"`
}

// QuoteLine is one proration line of a switch quote.
type QuoteLine struct {
        Description string `json:"description"`
        Amount      int64  `json:"amount"` // minor units; negative is a credit
}

// Quote tells a user what a purchase or plan change will cost before they commit to it. Amounts
// are in the currency's minor units (cents).
type Quote struct {
        Action         string        `json:"action"`
        CurrentProduct *QuoteProduct `json:"current_product,omitempty"`
        NewProduct     *QuoteProduct `json:"new_product,omitempty"`
        Quantity       int64         `json:"quantity"`
        Currency       string        `json:"currency,omitempty"`
        // AmountDueNow is what is charged when the user goes ahead: price x quantity for a checkout, the
        // prorated difference for an upgrade, and 0 for a downgrade.
        AmountDueNow int64 `json:"amount_due_now"`
        // CreditOnNextInvoice is the credit a downgrade earns against the next invoice.
        CreditOnNextInvoice int64       `json:"credit_on_next_invoice,omitempty"`
        ProrationLines      []QuoteLine `json:"proration_lines,omitempty"`
        // NextBillingDate is when the subscription renews next; empty for a one-off purchase.
        NextBillingDate *time.Time `json:"next_billing_date,omitempty"`
        // ProrationDate pins the proration to the moment of the quote. Send it back with confirm so the
        // charge matches the figure shown; omit it and Stripe prorates at the time of the request.
        ProrationDate int64 `json:"proration_date,omitempty"`
}

// ConfirmationRequiredError means a plan switch needs the user's explicit go-ahead. Nothing at
// Stripe has been changed; Quote is what to show them.
type ConfirmationRequiredError struct{ Quote *Quote }

func (e *ConfirmationRequiredError) Error() string { return "plan change needs confirmation" }

// Quote prices what Checkout would do for the same request, changing nothing. A product the user
// already holds is action "none". Every refusal Checkout makes (unknown product, no price, a
// quantity the product does not allow, over a maximum, nothing to raise) is returned the same way.
func (s *Service) Quote(ctx context.Context, userID string, req CheckoutRequest) (*Quote, error) {
        if req.ProductID == "" || req.TierID != "" {
                return nil, fmt.Errorf("%w: a quote needs a product_id", models.ErrInvalid)
        }
        var params CheckoutParams
        sw, err := s.productParams(ctx, userID, req, &params)
        if errors.Is(err, catalog.ErrInUse) {
                return s.heldQuote(ctx, req.ProductID)
        }
        if err != nil {
                return nil, err
        }
        if sw != nil {
                return s.switchQuote(ctx, userID, sw, params.PriceID, params.Quantity, req.ProrationDate)
        }
        return s.checkoutQuote(ctx, req.ProductID, params)
}

func (s *Service) productRef(ctx context.Context, id string) (models.Product, error) {
        products, err := s.store.GetProducts(ctx, []string{id})
        if err != nil || len(products) == 0 {
                return models.Product{}, fmt.Errorf("%w: product is not available", models.ErrInvalid)
        }
        return products[0], nil
}

func ref(p models.Product) *QuoteProduct { return &QuoteProduct{ID: p.Slug, Name: p.Name} }

func (s *Service) heldQuote(ctx context.Context, productID string) (*Quote, error) {
        p, err := s.productRef(ctx, productID)
        if err != nil {
                return nil, err
        }
        return &Quote{Action: QuoteNone, CurrentProduct: ref(p), NewProduct: ref(p), Quantity: 1}, nil
}

func (s *Service) checkoutQuote(ctx context.Context, productID string, params CheckoutParams) (*Quote, error) {
        p, err := s.productRef(ctx, productID)
        if err != nil {
                return nil, err
        }
        price, err := s.gw.Price(ctx, params.PriceID)
        if err != nil {
                return nil, providerFailed("quote", err, "product", productID, "price_id", params.PriceID)
        }
        qty := max(params.Quantity, 1)
        q := &Quote{
                Action: QuoteCheckout, NewProduct: ref(p), Quantity: qty,
                Currency: price.Currency, AmountDueNow: price.UnitAmount * qty,
        }
        if next, ok := nextPeriod(time.Now().UTC(), price); ok {
                q.NextBillingDate = &next
        }
        return q, nil
}

// nextPeriod is one billing interval after from, for a recurring price.
func nextPeriod(from time.Time, p PriceInfo) (time.Time, bool) {
        n := int(max(p.IntervalCount, 1))
        switch p.Interval {
        case "day":
                return from.AddDate(0, 0, n), true
        case "week":
                return from.AddDate(0, 0, 7*n), true
        case "month":
                return from.AddDate(0, n, 0), true
        case "year":
                return from.AddDate(n, 0, 0), true
        }
        return time.Time{}, false
}

// switchQuote previews the exact change a confirmed switch would make, using the proration and
// payment behavior decideSwitch chooses, so the figure is the one that will be charged.
func (s *Service) switchQuote(ctx context.Context, userID string, sw *planSwitch, priceID string, qty, prorationDate int64) (*Quote, error) {
        d, err := s.decideSwitch(ctx, userID, sw, priceID)
        if err != nil {
                return nil, err
        }
        if prorationDate == 0 {
                prorationDate = time.Now().Unix()
        }
        d.params.ProrationDate = prorationDate
        prev, err := s.gw.PreviewSwitch(ctx, d.params)
        if err != nil {
                return nil, providerFailed("quote", err, "product", sw.product.Slug, "subscription", sw.subscriptionID, "price_id", priceID)
        }

        q := &Quote{
                Action: QuoteSwitch, CurrentProduct: ref(sw.current), NewProduct: ref(sw.product),
                Quantity: d.params.Quantity, Currency: prev.Currency, ProrationDate: prorationDate,
        }
        if q.Currency == "" {
                q.Currency = d.price.Currency
        }
        next := d.sub.CurrentPeriodEnd
        if !next.IsZero() {
                q.NextBillingDate = &next
        }
        var sum int64
        for _, l := range prev.Lines {
                if l.Proration {
                        sum += l.Amount
                        q.ProrationLines = append(q.ProrationLines, QuoteLine{Description: l.Description, Amount: l.Amount})
                }
        }
        switch {
        case d.upgrade:
                q.AmountDueNow = max(sum, 0)
        case sum < 0:
                q.CreditOnNextInvoice = -sum
        }
        return q, nil
}

// CheckoutRequest names what to buy: exactly one of TierID or ProductID. Quantity applies to
// products and defaults to 1.
type CheckoutRequest struct {
        TierID    string
        ProductID string
        Quantity  int64
        // Confirm is the user's explicit go-ahead to switch an existing subscription to another plan.
        // Without it a switch returns ConfirmationRequiredError and changes nothing.
        Confirm bool
        // ProrationDate, from a quote, pins the proration so the charge matches the figure shown.
        ProrationDate int64
}

// CheckoutResult is either a Stripe-hosted checkout URL, or Switched when the user's existing
// subscription in the product's plan group was moved to the new price instead.
type CheckoutResult struct {
        URL      string
        Switched bool
}

// Checkout returns a Stripe-hosted checkout URL tied to userID. Anything not purchasable (no
// such tier or product, no Stripe price, a quantity the product does not allow) is rejected as
// invalid; a product the user already holds and cannot repeat, one that would pass a key's
// maximum, or one that would raise nothing, is refused before any Stripe session is created. A
// recurring product in a plan group the user already subscribes to switches that subscription
// to the new price (Stripe prorates) and returns Switched, creating no second subscription.
// email may be empty.
func (s *Service) Checkout(ctx context.Context, userID, email string, req CheckoutRequest) (CheckoutResult, error) {
        if (req.TierID == "") == (req.ProductID == "") {
                return CheckoutResult{}, fmt.Errorf("%w: provide either tier_id or product_id", models.ErrInvalid)
        }
        params := CheckoutParams{UserID: userID, Email: email, SuccessURL: s.urls.CheckoutSuccess, CancelURL: s.urls.CheckoutCancel}
        var (
                err error
                sw  *planSwitch
        )
        if req.TierID != "" {
                err = s.tierParams(ctx, req.TierID, &params)
        } else {
                sw, err = s.productParams(ctx, userID, req, &params)
        }
        if err != nil {
                return CheckoutResult{}, err
        }
        if sw != nil {
                return s.confirmedSwitch(ctx, userID, sw, params, req)
        }
        if params.CustomerID, err = s.store.StripeCustomer(ctx, userID); err != nil {
                return CheckoutResult{}, err
        }
        url, err := s.gw.CreateCheckout(ctx, params)
        if err != nil {
                return CheckoutResult{}, providerFailed("checkout", err, "product", req.ProductID, "tier", req.TierID, "price_id", params.PriceID)
        }
        return CheckoutResult{URL: url}, nil
}

// planSwitch is the subscription a checkout would move to another product in its plan group.
type planSwitch struct {
        subscriptionID string
        current        models.Product // what the subscription is on now
        product        models.Product // what it would move to
}

// maxProrationAge bounds a proration date taken from an earlier quote.
const maxProrationAge = time.Hour

// confirmedSwitch carries out a plan switch only once the user has confirmed it. Without
// confirmation it returns the quote to show them, and nothing at Stripe is changed.
func (s *Service) confirmedSwitch(ctx context.Context, userID string, sw *planSwitch, p CheckoutParams, req CheckoutRequest) (CheckoutResult, error) {
        if !req.Confirm {
                q, err := s.switchQuote(ctx, userID, sw, p.PriceID, p.Quantity, req.ProrationDate)
                if err != nil {
                        return CheckoutResult{}, err
                }
                return CheckoutResult{}, &ConfirmationRequiredError{Quote: q}
        }
        now := time.Now()
        if d := req.ProrationDate; d != 0 && (d > now.Unix() || time.Unix(d, 0).Before(now.Add(-maxProrationAge))) {
                return CheckoutResult{}, fmt.Errorf("%w: proration_date must be a time from a recent quote", models.ErrInvalid)
        }
        return s.switchPlan(ctx, userID, sw, p, req.ProrationDate)
}

// switchPlan is a decided plan change: what to send Stripe and whether it is an upgrade.
type switchPlan struct {
        params  SwitchParams
        upgrade bool
        sub     SubscriptionInfo
        price   PriceInfo
}

// decideSwitch reads the subscription and the new price from Stripe and chooses how to change
// plan. A change that raises the subscription's total (unit amount x quantity) is invoiced
// immediately and must be paid for; a change that lowers or keeps it credits the unused time to
// the next invoice.
func (s *Service) decideSwitch(ctx context.Context, userID string, sw *planSwitch, priceID string) (*switchPlan, error) {
        ids := []any{"product", sw.product.Slug, "subscription", sw.subscriptionID, "price_id", priceID}
        sub, err := s.gw.Subscription(ctx, sw.subscriptionID)
        if err != nil {
                return nil, providerFailed("plan switch", err, ids...)
        }
        price, err := s.gw.Price(ctx, priceID)
        if err != nil {
                return nil, providerFailed("plan switch", err, ids...)
        }
        if !strings.EqualFold(sub.Price.Currency, price.Currency) {
                return nil, fmt.Errorf("%w: the plans are billed in different currencies", models.ErrInvalid)
        }
        qty := max(sub.Quantity, 1)
        d := &switchPlan{sub: sub, price: price, upgrade: price.UnitAmount*qty > sub.Price.UnitAmount*qty}
        d.params = SwitchParams{
                SubscriptionID: sw.subscriptionID, CustomerID: sub.CustomerID, ItemID: sub.ItemID, Quantity: qty,
                PriceID: priceID, ProductID: sw.product.Slug, UserID: userID, Proration: ProrateCredit,
        }
        if d.upgrade {
                d.params.Proration, d.params.ChargeNow = ProrateInvoiceNow, true
        }
        return d, nil
}

// switchPlan changes the Stripe subscription and only then applies the new product's entitlements
// to its grants; the customer.subscription.updated event that follows finds nothing to change. A
// failed upgrade payment leaves both Stripe and the grants as they were.
func (s *Service) switchPlan(ctx context.Context, userID string, sw *planSwitch, p CheckoutParams, prorationDate int64) (CheckoutResult, error) {
        d, err := s.decideSwitch(ctx, userID, sw, p.PriceID)
        if err != nil {
                return CheckoutResult{}, err
        }
        d.params.ProrationDate = prorationDate
        if err := s.gw.SwitchPlan(ctx, d.params); err != nil {
                perr := providerFailed("plan switch", err, "product", sw.product.Slug, "subscription", sw.subscriptionID,
                        "price_id", p.PriceID, "proration", d.params.Proration)
                if pe, ok := perr.(*ProviderError); ok && d.params.ChargeNow && pe.PaymentFailed() {
                        return CheckoutResult{}, &PaymentFailedError{cause: pe}
                }
                return CheckoutResult{}, perr
        }
        grants, err := s.store.GrantsByExternal(ctx, models.SourceStripeSubscription, sw.subscriptionID)
        if err != nil {
                return CheckoutResult{}, err
        }
        return CheckoutResult{Switched: true}, s.replaceEntitlements(ctx, grants, sw.product, quantityOf(grants))
}

func (s *Service) tierParams(ctx context.Context, tierID string, p *CheckoutParams) error {
        tiers, err := s.store.GetTiers(ctx, []string{tierID})
        if err != nil {
                return err
        }
        if len(tiers) == 0 || tiers[0].StripePriceID == "" {
                return fmt.Errorf("%w: tier is not available for purchase", models.ErrInvalid)
        }
        p.PriceID, p.TierID = tiers[0].StripePriceID, tierID
        return nil
}

func (s *Service) productParams(ctx context.Context, userID string, req CheckoutRequest, p *CheckoutParams) (*planSwitch, error) {
        qty := req.Quantity
        if qty == 0 {
                qty = 1
        }
        if qty < 1 || qty > constants.MaxPurchaseQuantity {
                return nil, fmt.Errorf("%w: quantity must be between 1 and %d", models.ErrInvalid, constants.MaxPurchaseQuantity)
        }
        products, err := s.store.GetProducts(ctx, []string{req.ProductID})
        if err != nil {
                return nil, err
        }
        if len(products) == 0 || products[0].StripePriceID == "" {
                return nil, fmt.Errorf("%w: product is not available for purchase", models.ErrInvalid)
        }
        prod := products[0]
        if !prod.Repeatable {
                if qty != 1 {
                        return nil, fmt.Errorf("%w: this product cannot be bought with a quantity", models.ErrInvalid)
                }
                if err := s.checkNotHeld(ctx, userID, prod.Slug); err != nil {
                        return nil, err
                }
        }
        sw, err := s.findPlanMember(ctx, userID, prod)
        if err != nil {
                return nil, err
        }
        var opts []catalog.PurchaseOption
        if sw != nil {
                opts = append(opts, catalog.Replacing(sw.subscriptionID))
        }
        adds := make([]catalog.Units, 0, len(prod.Entitlements))
        for _, e := range prod.Entitlements {
                if e.Type == models.TypeLimit && !e.IsAtLeast() {
                        adds = append(adds, catalog.Units{Key: e.Key, Units: e.Value * qty})
                }
        }
        if err := s.limits.CheckPurchase(ctx, userID, adds, opts...); err != nil {
                return nil, err
        }
        if err := s.limits.CheckRaise(ctx, userID, prod.Entitlements, opts...); err != nil {
                return nil, err
        }
        p.PriceID, p.ProductID, p.Quantity = prod.StripePriceID, prod.Slug, qty
        if sw != nil {
                sw.product = prod
        }
        return sw, nil
}

// findPlanMember returns the user's active subscription to another product in prod's plan group,
// if any. Holding the same product again is refused. Products outside a group never switch.
func (s *Service) findPlanMember(ctx context.Context, userID string, prod models.Product) (*planSwitch, error) {
        if prod.PlanGroup == "" {
                return nil, nil
        }
        grants, err := s.store.ActiveGrants(ctx, userID, time.Now())
        if err != nil {
                return nil, err
        }
        var ids []string
        for _, g := range grants {
                if g.Source == models.SourceStripeSubscription && g.ProductID != "" && !slices.Contains(ids, g.ProductID) {
                        ids = append(ids, g.ProductID)
                }
        }
        if len(ids) == 0 {
                return nil, nil // no subscription to compare against
        }
        held, err := s.store.GetProducts(ctx, ids)
        if err != nil {
                return nil, err
        }
        for _, h := range held {
                if h.PlanGroup != prod.PlanGroup {
                        continue
                }
                if h.Slug == prod.Slug {
                        return nil, fmt.Errorf("%w: already subscribed to this product", catalog.ErrInUse)
                }
                for _, g := range grants {
                        if g.Source == models.SourceStripeSubscription && g.ProductID == h.Slug {
                                return &planSwitch{subscriptionID: g.ExternalID, current: h}, nil
                        }
                }
        }
        return nil, nil
}

// checkNotHeld refuses a second purchase of a non-repeatable product.
func (s *Service) checkNotHeld(ctx context.Context, userID, productID string) error {
        grants, err := s.store.ActiveGrants(ctx, userID, time.Now())
        if err != nil {
                return err
        }
        for _, g := range grants {
                if g.ProductID == productID {
                        return fmt.Errorf("%w: product already purchased", catalog.ErrInUse)
                }
        }
        return nil
}

        }
        return url, nil
}

// Event is a verified Stripe event reduced to what the handlers need.
type Event struct {
        ID     string
        Type   string
        Object json.RawMessage
}

// HandleEvent applies one Stripe event. Delivery is at-least-once, so it is idempotent twice
// over: an event ID already recorded is skipped, and every handler converges on the same grant
// state when replayed (the ID is recorded only after the handler succeeds, so a failed event is
// retried). Event types the service does not act on are ignored without being recorded.
func (s *Service) HandleEvent(ctx context.Context, ev Event) error {
        handle, ok := s.handlers()[ev.Type]
        if !ok {
                return nil
        }
        done, err := s.store.EventProcessed(ctx, ev.ID)
        if err != nil || done {
                return err
        }
        if err := handle(ctx, ev.Object); err != nil {
                return err
        }
        return s.store.RecordEvent(ctx, ev.ID, ev.Type, time.Now().UTC())
}

type handlerFunc func(ctx context.Context, object json.RawMessage) error

func (s *Service) handlers() map[string]handlerFunc {
        return map[string]handlerFunc{
                "checkout.session.completed":               s.onCheckout,
                "checkout.session.async_payment_succeeded": s.onCheckout,
                "invoice.paid":                             s.onInvoicePaid,
                "customer.subscription.updated":            s.onSubscriptionUpdated,
                "customer.subscription.deleted":            s.onSubscriptionDeleted,
                "charge.refunded":                          s.onChargeRefunded,
                "charge.dispute.created":                   s.onDispute,
        }
}

// purchase is what a Stripe event says a user bought: one Stripe object (the subscription or the
// payment intent) that becomes one grant per entitlement of a product, or one tier grant.
type purchase struct {
        source     string
        externalID string
        userID     string
        tierID     string
        productID  string
        quantity   int64
        validUntil *time.Time
        refs       []string
}

func (s *Service) onCheckout(ctx context.Context, raw json.RawMessage) error {
        var cs checkoutSession
        if err := json.Unmarshal(raw, &cs); err != nil {
                return err
        }
        if cs.PaymentStatus != "paid" && cs.PaymentStatus != "no_payment_required" {
                return nil // async payment pending; async_payment_succeeded follows
        }
        user := cs.Metadata["user_id"]
        if user == "" {
                user = cs.ClientReferenceID
        }
        if user == "" {
                logging.Logger.Warn("stripe checkout session has no user, ignoring", "session", cs.ID)
                return nil
        }
        if cs.Customer != "" {
                if err := s.store.SaveStripeCustomer(ctx, user, cs.Customer, systemActor); err != nil {
                        return err
                }
        }
        qty, _ := strconv.ParseInt(cs.Metadata["quantity"], 10, 64)
        p := purchase{
                userID: user, tierID: cs.Metadata["tier_id"], productID: cs.Metadata["product_id"],
                quantity: max(qty, 1), refs: nonEmpty(cs.ID),
        }
        switch cs.Mode {
        case "subscription":
                p.source, p.externalID = models.SourceStripeSubscription, cs.Subscription
        case "payment":
                p.source, p.externalID = models.SourceStripePurchase, cs.PaymentIntent
        default:
                return nil
        }
        if p.externalID == "" {
                return nil
        }
        return s.applyPurchase(ctx, p)
}

func (s *Service) onInvoicePaid(ctx context.Context, raw json.RawMessage) error {
        var inv invoice
        if err := json.Unmarshal(raw, &inv); err != nil {
                return err
        }
        sub, meta := inv.details()
        if sub == "" {
                return nil // not a subscription invoice
        }
        p := purchase{
                source: models.SourceStripeSubscription, externalID: sub,
                userID: meta["user_id"], tierID: meta["tier_id"], productID: meta["product_id"],
                quantity: inv.quantity(), refs: nonEmpty(inv.ID, inv.PaymentIntent, inv.Charge),
        }
        if end := inv.periodEnd(); end > 0 {
                t := time.Unix(end, 0).UTC()
                p.validUntil = &t
        }
        return s.applyPurchase(ctx, p)
}

// onSubscriptionUpdated ends the grants when Stripe gives up on a subscription, and follows a
// quantity change. past_due is left active: Stripe is still retrying the payment.
func (s *Service) onSubscriptionUpdated(ctx context.Context, raw json.RawMessage) error {
        var sub subscription
        if err := json.Unmarshal(raw, &sub); err != nil {
                return err
        }
        switch sub.Status {
        case "unpaid", "canceled", "incomplete_expired":
                return s.endGrants(ctx, models.SourceStripeSubscription, sub.ID)
        }
        return s.followSubscription(ctx, sub)
}

func (s *Service) onSubscriptionDeleted(ctx context.Context, raw json.RawMessage) error {
        var sub subscription
        if err := json.Unmarshal(raw, &sub); err != nil {
                return err
        }
        return s.endGrants(ctx, models.SourceStripeSubscription, sub.ID)
}

// onChargeRefunded acts only on a full refund; a partial refund leaves everything alone.
func (s *Service) onChargeRefunded(ctx context.Context, raw json.RawMessage) error {
        var ch charge
        if err := json.Unmarshal(raw, &ch); err != nil {
                return err
        }
        if !ch.Refunded {
                return nil
        }
        return s.settleReversal(ctx, "refund", ch.Invoice, ch.ID, ch.PaymentIntent, ch.Invoice)
}

// onDispute always revokes, and cancels a subscription, whatever the disputed charge was for.
func (s *Service) onDispute(ctx context.Context, raw json.RawMessage) error {
        var d dispute
        if err := json.Unmarshal(raw, &d); err != nil {
                return err
        }
        return s.settleReversal(ctx, "dispute", "", d.Charge, d.PaymentIntent)
}

// settleReversal revokes the grants a refund or dispute names. A one-off purchase is only revoked.
// For a subscription it also cancels the Stripe subscription, so a customer who got their money
// back is not billed again. A refund counts only when it is for the plan's own payment (its
// creation or a renewal); a refund of a plan-change charge is treated like a partial refund and
// changes nothing. refs are the Stripe object IDs the event names; invoiceID is the refunded
// charge's invoice, used only for refunds.
//
// Revoking comes first so the entitlement goes at once; the cancel that follows is idempotent
// (an already-canceled subscription is success), and a failed cancel returns an error so Stripe
// redelivers the event and the cancel is tried again.
func (s *Service) settleReversal(ctx context.Context, kind, invoiceID string, refs ...string) error {
        grants, err := s.grantsForRefs(ctx, refs)
        if err != nil {
                return err
        }
        var subscriptions []models.Grant
        for _, g := range grants {
                if g.Source == models.SourceStripeSubscription {
                        subscriptions = append(subscriptions, g)
                        continue
                }
                if err := s.revokeGrant(ctx, g); err != nil {
                        return err
                }
        }
        if len(subscriptions) == 0 {
                return nil
        }
        if kind == "refund" {
                if covers, err := s.refundCoversThePlan(ctx, invoiceID); err != nil || !covers {
                        return err
                }
        }
        for _, g := range subscriptions {
                if err := s.revokeGrant(ctx, g); err != nil {
                        return err
                }
        }
        for _, id := range distinctExternalIDs(subscriptions) {
                if err := s.cancelSubscription(ctx, id); err != nil {
                        return err
                }
        }
        return nil
}

// grantsForRefs returns every grant, in any status, that names one of the Stripe objects, once
// each. Revoked ones are included so a redelivery still reaches the cancel.
func (s *Service) grantsForRefs(ctx context.Context, refs []string) ([]models.Grant, error) {
        seen := map[string]bool{}
        var out []models.Grant
        for _, ref := range nonEmpty(refs...) {
                grants, err := s.store.GrantsByRef(ctx, ref)
                if err != nil {
                        return nil, err
                }
                for _, g := range grants {
                        if key := g.Source + "/" + g.ExternalID + "/" + g.Key + "/" + g.TierID; !seen[key] {
                                seen[key] = true
                                out = append(out, g)
                        }
                }
        }
        return out, nil
}

func (s *Service) revokeGrant(ctx context.Context, g models.Grant) error {
        if g.Status == models.GrantRevoked {
                return nil
        }
        g.Status = models.GrantRevoked
        return s.store.UpdateGrant(ctx, &g, systemActor)
}

func distinctExternalIDs(grants []models.Grant) []string {
        var ids []string
        for _, g := range grants {
                if !slices.Contains(ids, g.ExternalID) {
                        ids = append(ids, g.ExternalID)
                }
        }
        return ids
}

// planPaymentReasons are the invoice billing reasons for the plan's own payments.
var planPaymentReasons = []string{"subscription_create", "subscription_cycle"}

// refundCoversThePlan decides from the refunded charge's invoice whether a refund is for the
// plan itself. A refund with no invoice reference, or one whose invoice cannot be found, changes
// nothing and is logged; a failed lookup is returned so Stripe redelivers the event.
func (s *Service) refundCoversThePlan(ctx context.Context, invoiceID string) (bool, error) {
        if invoiceID == "" {
                logging.Logger.Warn("refunded charge names no invoice; leaving the subscription and grants as they are")
                return false, nil
        }
        inv, err := s.gw.Invoice(ctx, invoiceID)
        if err != nil {
                var se *stripe.Error
                if errors.As(err, &se) && se.Code == stripe.ErrorCodeResourceMissing {
                        logging.Logger.Warn("refunded charge's invoice was not found; leaving the subscription and grants as they are", "invoice", invoiceID)
                        return false, nil
                }
                return false, providerFailed("refund invoice lookup", err, "invoice", invoiceID)
        }
        if !slices.Contains(planPaymentReasons, inv.BillingReason) {
                logging.Logger.Info("refund is for a plan-change charge, not the plan's own payment; keeping the subscription and grants",
                        "invoice", invoiceID, "billing_reason", inv.BillingReason)
                return false, nil
        }
        return true, nil
}

// cancelSubscription cancels the Stripe subscription now. One that is already canceled or gone is
// the state wanted, so it is success.
func (s *Service) cancelSubscription(ctx context.Context, subscriptionID string) error {
        if err := s.gw.CancelSubscription(ctx, subscriptionID); err != nil && !alreadyCanceled(err) {
                return providerFailed("cancel subscription", err, "subscription", subscriptionID)
        }
        logging.Logger.Info("subscription canceled after a refund or dispute", "subscription", subscriptionID)
        return nil
}

// wanted expands a purchase into the grants it should have: one key-level grant per entitlement
// of a product (value x quantity; a flag is simply on), or a single tier grant. A product that
// no longer exists yields none.
func (s *Service) wanted(ctx context.Context, p purchase) ([]models.Grant, error) {
        base := models.Grant{
                UserID: p.userID, Source: p.source, ExternalID: p.externalID,
                ValidUntil: p.validUntil, Refs: p.refs, Status: models.GrantActive,
        }
        switch {
        case p.productID != "":
                products, err := s.store.GetProducts(ctx, []string{p.productID})
                if err != nil {
                        return nil, err
                }
                if len(products) == 0 {
                        logging.Logger.Warn("stripe event names a product that is not in the catalog", "product", p.productID, "external_id", p.externalID)
                        return nil, nil
                }
                out := make([]models.Grant, 0, len(products[0].Entitlements))
                for _, e := range products[0].Entitlements {
                        g := base
                        g.Key, g.Value, g.ProductID, g.Quantity, g.Mode = e.Key, grantValue(e, p.quantity), p.productID, p.quantity, e.Mode
                        out = append(out, g)
                }
                return out, nil
        case p.tierID != "":
                g := base
                g.TierID = p.tierID
                return []models.Grant{g}, nil
        }
        return nil, nil
}

// grantValue is what one purchase grants: a flag is on, an at_least floor is the same whatever the
// quantity, and an add entitlement is its value per unit x quantity.
func grantValue(e models.ProductEntitlement, quantity int64) int64 {
        if e.Type == models.TypeFlag {
                return 1
        }
        if e.IsAtLeast() {
                return e.Value
        }
        return e.Value * quantity
}

// applyPurchase creates the grants a purchase should have and folds a later event into the ones
// that exist: it reactivates an ended grant, only ever extends valid_until, and unions refs. A
// revoked grant is never touched. Existing grants are updated even when the event carries no
// product or tier (a renewal invoice), and an event that would create grants without knowing
// the user is ignored.
func (s *Service) applyPurchase(ctx context.Context, p purchase) error {
        existing, err := s.store.GrantsByExternal(ctx, p.source, p.externalID)
        if err != nil {
                return err
        }
        want, err := s.wanted(ctx, p)
        if err != nil {
                return err
        }
        for i := range existing {
                if err := s.mergeGrant(ctx, &existing[i], p); err != nil {
                        return err
                }
        }
        for _, g := range want {
                if slices.ContainsFunc(existing, func(e models.Grant) bool { return e.Key == g.Key && e.TierID == g.TierID }) {
                        continue
                }
                if g.UserID == "" {
                        logging.Logger.Warn("stripe event cannot create a grant without a user", "external_id", p.externalID)
                        return nil
                }
                if err := s.store.InsertGrant(ctx, &g, systemActor); err != nil {
                        return err
                }
        }
        return nil
}

func (s *Service) mergeGrant(ctx context.Context, cur *models.Grant, p purchase) error {
        if cur.Status == models.GrantRevoked {
                return nil
        }
        cur.Status = models.GrantActive
        if p.userID != "" {
                cur.UserID = p.userID
        }
        if p.validUntil != nil && (cur.ValidUntil == nil || p.validUntil.After(*cur.ValidUntil)) {
                cur.ValidUntil = p.validUntil
        }
        for _, r := range p.refs {
                if !slices.Contains(cur.Refs, r) {
                        cur.Refs = append(cur.Refs, r)
                }
        }
        return s.store.UpdateGrant(ctx, cur, systemActor)
}

// followSubscription reconciles a subscription's grants with its current price and quantity. The
// price decides the product (a customer who changed plan in the billing portal changes the
// price, not our metadata); metadata is the fallback. Stripe has already accepted the change, so
// a result above a key's maximum cannot be refused here; it is logged and checks clamp it.
func (s *Service) followSubscription(ctx context.Context, sub subscription) error {
        qty := sub.quantity()
        if qty < 1 {
                return nil
        }
        prod, err := s.productFor(ctx, sub)
        if err != nil || prod == nil {
                return err
        }
        grants, err := s.store.GrantsByExternal(ctx, models.SourceStripeSubscription, sub.ID)
        if err != nil || len(grants) == 0 {
                return err
        }
        return s.replaceEntitlements(ctx, grants, *prod, qty)
}

func (s *Service) productFor(ctx context.Context, sub subscription) (*models.Product, error) {
        if price := sub.priceID(); price != "" {
                p, err := s.store.ProductByPriceID(ctx, price)
                if err == nil {
                        return p, nil
                }
                if !errors.Is(err, data.ErrNotFound) {
                        return nil, err
                }
        }
        if id := sub.Metadata["product_id"]; id != "" {
                products, err := s.store.GetProducts(ctx, []string{id})
                if err != nil || len(products) == 0 {
                        return nil, err
                }
                return &products[0], nil
        }
        return nil, nil
}

// replaceEntitlements makes a purchase's grants match a product at a quantity: a grant for each
// of the product's keys (updated in place, or created), and any grant for a key it no longer
// has ended. It keeps one set of grants per Stripe object, sharing its ID and validity. A revoked
// purchase is left alone.
func (s *Service) replaceEntitlements(ctx context.Context, grants []models.Grant, prod models.Product, qty int64) error {
        if len(grants) == 0 || slices.ContainsFunc(grants, func(g models.Grant) bool { return g.Status == models.GrantRevoked }) {
                return nil
        }
        keep := map[string]bool{}
        for _, e := range prod.Entitlements {
                keep[e.Key] = true
                want := grantValue(e, qty)
                i := slices.IndexFunc(grants, func(g models.Grant) bool { return g.Key == e.Key })
                if i < 0 {
                        g := grants[0]
                        g.ID, g.Auditable, g.TierID, g.Status = primitive.NilObjectID, mcore.Auditable{}, "", models.GrantActive
                        g.Key, g.Value, g.ProductID, g.Quantity, g.Mode = e.Key, want, prod.Slug, qty, e.Mode
                        if err := s.store.InsertGrant(ctx, &g, systemActor); err != nil {
                                return err
                        }
                        continue
                }
                g := &grants[i]
                if g.Value == want && g.Quantity == qty && g.ProductID == prod.Slug && g.Mode == e.Mode && g.Status == models.GrantActive {
                        continue
                }
                g.Value, g.Quantity, g.ProductID, g.Mode, g.Status = want, qty, prod.Slug, e.Mode, models.GrantActive
                if err := s.store.UpdateGrant(ctx, g, systemActor); err != nil {
                        return err
                }
                s.warnIfOverMaximum(ctx, g)
        }
        for i := range grants {
                if keep[grants[i].Key] || grants[i].Status != models.GrantActive {
                        continue
                }
                grants[i].Status = models.GrantEnded
                if err := s.store.UpdateGrant(ctx, &grants[i], systemActor); err != nil {
                        return err
                }
        }
        return nil
}

// quantityOf is the quantity a purchase was made at, 1 if unrecorded.
func quantityOf(grants []models.Grant) int64 {
        for _, g := range grants {
                if g.Quantity > 0 {
                        return g.Quantity
                }
        }
        return 1
}

// warnIfOverMaximum probes whether the user now holds more than the key's maximum.
func (s *Service) warnIfOverMaximum(ctx context.Context, g *models.Grant) {
        var exceeded catalog.MaxExceededError
        if err := s.limits.CheckPurchase(ctx, g.UserID, []catalog.Units{{Key: g.Key}}); errors.As(err, &exceeded) {
                logging.Logger.Warn("subscription quantity change passes the key maximum; checks clamp it",
                        "user", g.UserID, "key", g.Key, "external_id", g.ExternalID)
        }
}

// endGrants ends every active grant of one Stripe object together.
func (s *Service) endGrants(ctx context.Context, source, externalID string) error {
        grants, err := s.store.GrantsByExternal(ctx, source, externalID)
        if err != nil {
                return err
        }
        for i := range grants {
                if grants[i].Status != models.GrantActive {
                        continue
                }
                grants[i].Status = models.GrantEnded
                if err := s.store.UpdateGrant(ctx, &grants[i], systemActor); err != nil {
                        return err
                }
        }
        return nil
}

        return slices.DeleteFunc(slices.Clone(vals), func(v string) bool { return v == "" })
}

// Result is one key's effective value for a user. Usage is never included; counting against a
// limit is the consumer's job.
type Result struct {
        Key        string
        Type       models.KeyType
        Value      int64
        TTLSeconds int
}

// UnknownKeysError lists requested keys that are not in the catalog.
type UnknownKeysError struct{ Keys []string }

func (e UnknownKeysError) Error() string { return fmt.Sprintf("unknown keys: %v", e.Keys) }

// Check returns the effective value of each named key for userID, in the order given. Any key
// missing from the catalog fails the whole call with UnknownKeysError rather than defaulting.
func (s *Service) Check(ctx context.Context, userID string, names []string) ([]Result, error) {
        names = dedupe(names)
        keys, err := s.store.GetKeys(ctx, names)
        if err != nil {
                return nil, err
        }
        return s.resolveKeys(ctx, userID, names, keys)
}

// CheckPrefix returns every live key under prefix (up to MaxBatchKeys) with its effective value.
func (s *Service) CheckPrefix(ctx context.Context, userID, prefix string) ([]Result, error) {
        keys, err := s.store.ListKeys(ctx, prefix, constants.MaxBatchKeys, 0)
        if err != nil {
                return nil, err
        }
        names := make([]string, len(keys))
        for i, k := range keys {
                names[i] = k.Key
        }
        return s.resolveKeys(ctx, userID, names, keys)
}

// resolveKeys returns clamped effective values: a limit key with a maximum never reports more.
func (s *Service) resolveKeys(ctx context.Context, userID string, names []string, keys []models.Key) ([]Result, error) {
        byName, values, err := s.rawValues(ctx, userID, names, keys)
        if err != nil {
                return nil, err
        }
        out := make([]Result, 0, len(names))
        for _, n := range names {
                k := byName[n]
                out = append(out, Result{Key: n, Type: k.Type, Value: clamp(k, values[n]), TTLSeconds: k.TTLSeconds})
        }
        return out, nil
}

func clamp(k models.Key, v int64) int64 {
        if k.Max != nil && v > *k.Max {
                return *k.Max
        }
        return v
}

// rawValues resolves the unclamped effective value of each key. Any name missing from keys fails
// the call with UnknownKeysError.
func (s *Service) rawValues(ctx context.Context, userID string, names []string, keys []models.Key) (map[string]models.Key, map[string]int64, error) {
        return s.rawValuesWith(ctx, userID, names, keys, nil, "")
}

// rawValuesWith is rawValues with extra hypothetical grants counted and the grants of one Stripe
// object (replacing, if set) left out, to project a purchase.
func (s *Service) rawValuesWith(ctx context.Context, userID string, names []string, keys []models.Key, extra []models.Grant, replacing string) (map[string]models.Key, map[string]int64, error) {
        byName := make(map[string]models.Key, len(keys))
        for _, k := range keys {
                byName[k.Key] = k
        }
        var unknown []string
        for _, n := range names {
                if _, ok := byName[n]; !ok {
                        unknown = append(unknown, n)
                }
        }
        if len(unknown) > 0 {
                return nil, nil, UnknownKeysError{Keys: unknown}
        }

        now := s.now().UTC()
        grants, err := s.store.ActiveGrants(ctx, userID, now)
        if err != nil {
                return nil, nil, err
        }
        if replacing != "" {
                grants = slices.DeleteFunc(grants, func(g models.Grant) bool { return g.ExternalID == replacing })
        }
        grants = append(grants, extra...)
        tiers, err := s.loadTiers(ctx, grants)
        if err != nil {
                return nil, nil, err
        }

        types := make(map[string]models.KeyType, len(byName))
        for n, k := range byName {
                types[n] = k.Type
        }
        return byName, Resolve(tiers.def, tiers.granted, grants, types, now), nil
}

type loadedTiers struct {
        def     *models.Tier
        granted map[string]models.Tier
}

// loadTiers fetches the default tier and every tier the grants reference in one query.
func (s *Service) loadTiers(ctx context.Context, grants []models.Grant) (loadedTiers, error) {
        defID, err := s.store.DefaultTierID(ctx)
        if err != nil {
                return loadedTiers{}, err
        }
        ids := []string{}
        if defID != "" {
                ids = append(ids, defID)
        }
        for _, g := range grants {
                if g.TierID != "" && !slices.Contains(ids, g.TierID) {
                        ids = append(ids, g.TierID)
                }
        }
        found, err := s.store.GetTiers(ctx, ids)
        if err != nil {
                return loadedTiers{}, err
        }
        lt := loadedTiers{granted: map[string]models.Tier{}}
        for _, t := range found {
                lt.granted[t.Slug] = t
                if t.Slug == defID {
                        c := t
                        lt.def = &c
                }
        }
        return lt, nil
}

func dedupe(in []string) []string {
        seen := make(map[string]bool, len(in))
        out := make([]string, 0, len(in))
        for _, s := range in {
                if !seen[s] {
                        seen[s] = true
                        out = append(out, s)
                }
        }
        return out
}

// ProductInput is the writable part of a product. Entitlements maps key name to a JSON bool or
// number: the value granted per unit purchased.
type ProductInput struct {
        Slug          string
        Name          string
        Description   string
        Billing       string
        Repeatable    bool
        PlanGroup     string
        StripePriceID string
        Entitlements  map[string]any
}

// buildEntitlements validates every key exists and the value matches its type. A limit
// entitlement must add at least one unit; a flag entitlement must be true.
func (s *Service) buildEntitlements(ctx context.Context, raw map[string]any) ([]models.ProductEntitlement, error) {
        if len(raw) == 0 {
                return nil, fmt.Errorf("%w: a product needs at least one entitlement", models.ErrInvalid)
        }
        plain, modes, err := splitModes(raw)
        if err != nil {
                return nil, err
        }
        values, err := s.buildValues(ctx, plain)
        if err != nil {
                return nil, err
        }
        out := make([]models.ProductEntitlement, len(values))
        for i, v := range values {
                if v.Value < 1 {
                        return nil, fmt.Errorf("%w: entitlement %q must grant at least 1", models.ErrInvalid, v.Key)
                }
                out[i] = models.ProductEntitlement{Key: v.Key, Type: v.Type, Value: v.Value, Mode: modes[v.Key]}
                if out[i].Mode == models.ModeAtLeast && v.Type != models.TypeLimit {
                        return nil, fmt.Errorf("%w: at_least applies only to limit keys (%q)", models.ErrInvalid, v.Key)
                }
        }
        return out, nil
}

// splitModes separates each entitlement's value from its mode. An entitlement is either a bare
// value (mode add) or {"value": v, "mode": "add"|"at_least"}.
func splitModes(raw map[string]any) (map[string]any, map[string]string, error) {
        plain := make(map[string]any, len(raw))
        modes := make(map[string]string, len(raw))
        for k, v := range raw {
                obj, isObj := v.(map[string]any)
                if !isObj {
                        plain[k], modes[k] = v, models.ModeAdd
                        continue
                }
                mode, _ := obj["mode"].(string)
                if mode == "" {
                        mode = models.ModeAdd
                }
                if mode != models.ModeAdd && mode != models.ModeAtLeast {
                        return nil, nil, fmt.Errorf("%w: mode for %q must be add or at_least", models.ErrInvalid, k)
                }
                plain[k], modes[k] = obj["value"], mode
        }
        return plain, modes, nil
}

func (s *Service) productFromInput(ctx context.Context, in ProductInput) (*models.Product, error) {
        if err := models.ValidateProductSlug(in.Slug); err != nil {
                return nil, err
        }
        if strings.TrimSpace(in.Name) == "" {
                return nil, fmt.Errorf("%w: name is required", models.ErrInvalid)
        }
        if !slices.Contains([]string{models.BillingOneOff, models.BillingRecurring}, in.Billing) {
                return nil, fmt.Errorf("%w: billing must be one_off or recurring", models.ErrInvalid)
        }
        if in.PlanGroup != "" {
                if in.Billing != models.BillingRecurring {
                        return nil, fmt.Errorf("%w: only recurring products can be in a plan group", models.ErrInvalid)
                }
                if err := models.ValidateTierSlug(in.PlanGroup); err != nil {
                        return nil, fmt.Errorf("%w: plan_group must be lowercase letters, digits, - or _ (max 64)", models.ErrInvalid)
                }
        }
        ents, err := s.buildEntitlements(ctx, in.Entitlements)
        if err != nil {
                return nil, err
        }
        return &models.Product{
                Slug: in.Slug, Name: in.Name, Description: in.Description, Billing: in.Billing,
                Repeatable: in.Repeatable, PlanGroup: in.PlanGroup, StripePriceID: in.StripePriceID, Entitlements: ents,
        }, nil
}

        } else if len(existing) > 0 {
                return nil, ErrDuplicate
        }
        if err := s.store.InsertProduct(ctx, p, actor); err != nil {
                return nil, err
        }
        return p, s.audit(ctx, actor, "create", "product", p.Slug, nil, p)
}

// PurchaseOption adjusts how a purchase is checked.
type PurchaseOption func(*purchaseOptions)

type purchaseOptions struct{ replacing string }

// Replacing judges the purchase as if the grants of one Stripe object (a subscription being
// switched to another plan) were gone.
func Replacing(externalID string) PurchaseOption {
        return func(o *purchaseOptions) { o.replacing = externalID }
}

func applyOptions(opts []PurchaseOption) purchaseOptions {
        var o purchaseOptions
        for _, f := range opts {
                f(&o)
        }
        return o
}

// Units is a quantity of one limit key a purchase would add.
type Units struct {
        Key   string
        Units int64
}

// MaxExceededError means a purchase would take a key past its maximum. Remaining is how many
// units the user can still add.
type MaxExceededError struct {
        Key       string
        Remaining int64
}

func (e MaxExceededError) Error() string {
        return fmt.Sprintf("%s: purchase exceeds the maximum, %d units remaining", e.Key, e.Remaining)
}

// CheckPurchase refuses a purchase that would pass a limit key's maximum. The user's current
// holding is the unclamped effective value (default tier, tier grants, and purchased units), so a
// value already at or past the maximum leaves nothing to buy. Flags and keys without a maximum
// always pass.
func (s *Service) CheckPurchase(ctx context.Context, userID string, adds []Units, opts ...PurchaseOption) error {
        names := make([]string, len(adds))
        for i, a := range adds {
                names[i] = a.Key
        }
        names = dedupe(names)
        keys, err := s.store.GetKeys(ctx, names)
        if err != nil {
                return err
        }
        byName, values, err := s.rawValuesWith(ctx, userID, names, keys, nil, applyOptions(opts).replacing)
        if err != nil {
                return err
        }
        total := map[string]int64{}
        for _, a := range adds {
                total[a.Key] += a.Units
        }
        for _, n := range names {
                k := byName[n]
                if k.Type != models.TypeLimit || k.Max == nil {
                        continue
                }
                if values[n]+total[n] > *k.Max {
                        return MaxExceededError{Key: n, Remaining: max(0, *k.Max-values[n])}
                }
        }
        return nil
}

// NoIncreaseError means a product's at_least entitlements would not raise the user's value for
// any of their keys, so buying it would change nothing.
type NoIncreaseError struct{ Keys []string }

func (e NoIncreaseError) Error() string {
        return fmt.Sprintf("purchase would not raise %v: already at or above it", e.Keys)
}

// CheckRaise refuses a purchase whose at_least entitlements would not raise the user's current
// (clamped) value. A bundle passes when any of its at_least entitlements raises one; a product
// with no at_least entitlement always passes.
func (s *Service) CheckRaise(ctx context.Context, userID string, ents []models.ProductEntitlement, opts ...PurchaseOption) error {
        replacing := applyOptions(opts).replacing
        var atLeast []models.ProductEntitlement
        var names []string
        for _, e := range ents {
                if e.IsAtLeast() && e.Type == models.TypeLimit {
                        atLeast = append(atLeast, e)
                        names = append(names, e.Key)
                }
        }
        if len(atLeast) == 0 {
                return nil
        }
        names = dedupe(names)
        keys, err := s.store.GetKeys(ctx, names)
        if err != nil {
                return err
        }
        byName, now, err := s.rawValuesWith(ctx, userID, names, keys, nil, replacing)
        if err != nil {
                return err
        }
        hypothetical := make([]models.Grant, len(atLeast))
        for i, e := range atLeast {
                hypothetical[i] = models.Grant{UserID: userID, Key: e.Key, Value: e.Value, Mode: models.ModeAtLeast, Status: models.GrantActive}
        }
        _, projected, err := s.rawValuesWith(ctx, userID, names, keys, hypothetical, replacing)
        if err != nil {
                return err
        }
        for _, n := range names {
                if clamp(byName[n], projected[n]) > clamp(byName[n], now[n]) {
                        return nil
                }
        }
        return NoIncreaseError{Keys: names}
}

// Resolve returns the effective value per key:
//
//        effective = max(default tier, tier grants, at_least grants) + sum(add grants)
//
// for a limit key; a flag key is on when its tier-derived value is on or any key-level grant is
// active. Tiers and at_least grants are alternatives (maximum); purchased add units stack. types says which keys are
// flags, any key not listed is treated as a limit. A key with no source is absent from the map
// (read as zero or false). Grants are re-checked for activity at now so the rule holds for any
// Store.
func Resolve(def *models.Tier, tiers map[string]models.Tier, grants []models.Grant, types map[string]models.KeyType, now time.Time) map[string]int64 {
        tierValue := map[string]int64{}
        raise := func(key string, v int64) {
                if cur, ok := tierValue[key]; !ok || v > cur {
                        tierValue[key] = v
                }
        }
        if def != nil {
                for _, tv := range def.Values {
                        raise(tv.Key, tv.Value)
                }
        }
        units := map[string]int64{}
        for _, g := range grants {
                if !g.ActiveAt(now) {
                        continue
                }
                if g.TierID != "" {
                        for _, tv := range tiers[g.TierID].Values {
                                raise(tv.Key, tv.Value)
                        }
                }
                if g.Key == "" {
                        continue
                }
                switch {
                case types[g.Key] == models.TypeFlag:
                        units[g.Key] = 1
                case g.Mode == models.ModeAtLeast:
                        raise(g.Key, g.Value)
                default:
                        units[g.Key] += g.Value
                }
        }

        out := tierValue
        for key, add := range units {
                if types[key] == models.TypeFlag {
                        out[key] = max(out[key], add)
                } else {
                        out[key] += add
                }
        }
        return out
}

// Sentinel errors handlers map to HTTP statuses. Validation failures wrap models.ErrInvalid.
var (
        ErrNotFound  = data.ErrNotFound
        ErrDuplicate = data.ErrDuplicate
        // ErrInUse means the change would orphan something that still depends on the target.
        ErrInUse = errors.New("in use")
)

// Service owns catalog writes and checks.
type Service struct {
        store data.Store
        now   func() time.Time
}

// New returns a Service over store.
func New(store data.Store) *Service { return &Service{store: store, now: time.Now} }

// KeyInput is the writable part of a key.
type KeyInput struct {
        Key         string
        Type        models.KeyType
        Description string
        TTLSeconds  int
        // Max caps a limit key's effective value; nil means no maximum.
        Max *int64
}

// TierInput is the writable part of a tier. Values maps key name to a JSON bool or number.
type TierInput struct {
        Slug          string
        Name          string
        Description   string
        Values        map[string]any
        StripePriceID string
}

func validateTTL(ttl int) error {
        if ttl < 0 || ttl > constants.MaxTTLSeconds {
                return fmt.Errorf("%w: ttl_seconds must be between 0 and %d", models.ErrInvalid, constants.MaxTTLSeconds)
        }
        return nil
}

// audit appends the trail entry after a successful write. The write and the entry are not one
// transaction: if the append fails the change is already persisted, so the failure is logged at
// ERROR and the write still reports success. Upgrade path: wrap both in a Mongo transaction if
// lost entries ever matter.
func (s *Service) audit(ctx context.Context, actor, action, kind, subject string, before, after any) error {
        err := s.store.AppendAudit(ctx, models.AuditEntry{
                At: s.now().UTC(), Actor: actor, Action: action, Kind: kind, Subject: subject, Before: before, After: after,
        })
        if err != nil {
                logging.Logger.Error("catalog audit entry not recorded", "action", action, "kind", kind, "subject", subject, "actor", actor, "error", err.Error())
        }
        return nil
}

// CreateKey registers a key. A live key with the same name is rejected.
func (s *Service) CreateKey(ctx context.Context, actor string, in KeyInput) (*models.Key, error) {
        if err := models.ValidateKeyName(in.Key); err != nil {
                return nil, err
        }
        if in.Type != models.TypeFlag && in.Type != models.TypeLimit {
                return nil, fmt.Errorf("%w: type must be flag or limit", models.ErrInvalid)
        }
        if err := validateTTL(in.TTLSeconds); err != nil {
                return nil, err
        }
        if err := models.ValidateMax(in.Type, in.Max); err != nil {
                return nil, err
        }
        if existing, err := s.store.GetKeys(ctx, []string{in.Key}); err != nil {
                return nil, err
        } else if len(existing) > 0 {
                return nil, ErrDuplicate
        }
        k := &models.Key{Key: in.Key, Type: in.Type, Description: in.Description, TTLSeconds: in.TTLSeconds, Max: in.Max}
        if err := s.store.InsertKey(ctx, k, actor); err != nil {
                return nil, err
        }
        return k, s.audit(ctx, actor, "create", "key", k.Key, nil, k)
}

// UpdateKey changes a key's description, TTL, and maximum (nil removes the maximum). Its type
// cannot change.
func (s *Service) UpdateKey(ctx context.Context, actor, name, description string, ttl int, max *int64) (*models.Key, error) {
        if err := validateTTL(ttl); err != nil {
                return nil, err
        }
        before, err := s.store.GetKeys(ctx, []string{name})
        if err != nil {
                return nil, err
        }
        if len(before) == 0 {
                return nil, ErrNotFound
        }
        if err := models.ValidateMax(before[0].Type, max); err != nil {
                return nil, err
        }
        k, err := s.store.UpdateKey(ctx, name, description, ttl, max, actor)
        if err != nil {
                return nil, err
        }
        return k, s.audit(ctx, actor, "update", "key", name, before[0], k)
}

// DeleteKey removes a key from the catalog. Tier values that still name it stay stored but are
// never returned, because a check resolves only live keys.
func (s *Service) DeleteKey(ctx context.Context, actor, name string) error {
        if err := s.store.DeleteKey(ctx, name, actor); err != nil {
                return err
        }
        return s.audit(ctx, actor, "delete", "key", name, nil, nil)
}

// ListKeys returns live keys under prefix, ordered by name.
func (s *Service) ListKeys(ctx context.Context, prefix string, limit, skip int64) ([]models.Key, error) {
        return s.store.ListKeys(ctx, prefix, limit, skip)
}

// buildValues validates every named key exists and the value matches its type.
func (s *Service) buildValues(ctx context.Context, raw map[string]any) ([]models.TierValue, error) {
        names := make([]string, 0, len(raw))
        for n := range raw {
                names = append(names, n)
        }
        slices.Sort(names)
        keys, err := s.store.GetKeys(ctx, names)
        if err != nil {
                return nil, err
        }
        byName := map[string]models.Key{}
        for _, k := range keys {
                byName[k.Key] = k
        }
        out := make([]models.TierValue, 0, len(names))
        for _, n := range names {
                k, ok := byName[n]
                if !ok {
                        return nil, fmt.Errorf("%w: unknown key %q", models.ErrInvalid, n)
                }
                v, err := models.ParseValue(k.Type, raw[n])
                if err != nil {
                        return nil, fmt.Errorf("%w (key %q)", err, n)
                }
                out = append(out, models.TierValue{Key: n, Type: k.Type, Value: v})
        }
        return out, nil
}

func (s *Service) tierFromInput(ctx context.Context, in TierInput) (*models.Tier, error) {
        if err := models.ValidateTierSlug(in.Slug); err != nil {
                return nil, err
        }
        if strings.TrimSpace(in.Name) == "" {
                return nil, fmt.Errorf("%w: name is required", models.ErrInvalid)
        }
        values, err := s.buildValues(ctx, in.Values)
        if err != nil {
                return nil, err
        }
        return &models.Tier{Slug: in.Slug, Name: in.Name, Description: in.Description, Values: values, StripePriceID: in.StripePriceID}, nil
}

// CreateTier adds a tier.
func (s *Service) CreateTier(ctx context.Context, actor string, in TierInput) (*models.Tier, error) {
        t, err := s.tierFromInput(ctx, in)
        if err != nil {
                return nil, err
        }
        if existing, err := s.store.GetTiers(ctx, []string{t.Slug}); err != nil {
                return nil, err
        } else if len(existing) > 0 {
                return nil, ErrDuplicate
        }
        if err := s.store.InsertTier(ctx, t, actor); err != nil {
                return nil, err
        }
        return t, s.audit(ctx, actor, "create", "tier", t.Slug, nil, t)
}

// UpdateTier replaces a tier's name, description, values, and Stripe price ID.
func (s *Service) UpdateTier(ctx context.Context, actor string, in TierInput) (*models.Tier, error) {
        t, err := s.tierFromInput(ctx, in)
        if err != nil {
                return nil, err
        }
        before, err := s.store.GetTiers(ctx, []string{in.Slug})
        if err != nil {
                return nil, err
        }
        if len(before) == 0 {
                return nil, ErrNotFound
        }
        updated, err := s.store.UpdateTier(ctx, t, actor)
        if err != nil {
                return nil, err
        }
        return updated, s.audit(ctx, actor, "update", "tier", in.Slug, before[0], updated)
}

// DeleteTier removes a tier. The default tier cannot be deleted; pick another default first.
func (s *Service) DeleteTier(ctx context.Context, actor, slug string) error {
        def, err := s.store.DefaultTierID(ctx)
        if err != nil {
                return err
        }
        if def == slug {
                return fmt.Errorf("%w: tier is the default tier", ErrInUse)
        }
        if err := s.store.DeleteTier(ctx, slug, actor); err != nil {
                return err
        }
        return s.audit(ctx, actor, "delete", "tier", slug, nil, nil)
}

// ListTiers returns live tiers ordered by id.
func (s *Service) ListTiers(ctx context.Context, limit, skip int64) ([]models.Tier, error) {
        return s.store.ListTiers(ctx, limit, skip)
}

// SetDefaultTier makes an existing tier the one every user implicitly holds.
func (s *Service) SetDefaultTier(ctx context.Context, actor, slug string) error {
        tiers, err := s.store.GetTiers(ctx, []string{slug})
        if err != nil {
                return err
        }
        if len(tiers) == 0 {
                return ErrNotFound
        }
        before, err := s.store.DefaultTierID(ctx)
        if err != nil {
                return err
        }
        if err := s.store.SetDefaultTierID(ctx, slug, actor); err != nil {
                return err
        }
        return s.audit(ctx, actor, "set_default", "default_tier", slug, before, slug)
}

// DefaultTier returns the default tier id, or "" when none is configured.
func (s *Service) DefaultTier(ctx context.Context) (string, error) {
        return s.store.DefaultTierID(ctx)
}

// ListAudit returns audit entries newest first.
func (s *Service) ListAudit(ctx context.Context, limit, skip int64) ([]models.AuditEntry, error) {
        return s.store.ListAudit(ctx, limit, skip)
}

        if err != nil {
                logging.Logger.Error("Refusing to start", "error", err.Error())
                os.Exit(1)
        }

        setupSentry()

        ff := featureflags.New(constants.ServiceName)

                defer stopProfiling()
        }

        httpLogger := slog.New(slog.NewJSONHandler(os.Stdout, nil))

        r.Use(sloggin.New(httpLogger))
        r.Use(gin.Recovery())

        setupMetrics(r)

        database.SetupDatabase()
        defer database.TeardownDatabase()

        store := data.NewMongoStore(database.Db)
        if err := store.EnsureIndexes(context.Background()); err != nil {
                // Without its unique indexes the database does not enforce uniqueness, so refuse to serve.
                logging.Logger.Error("Failed to ensure indexes", "error", err.Error())
                os.Exit(1)
        }

        setupAcuator(r)

        setupSwagger(r)

        // Per-client/IP rate limiter (Redis-backed, fail-closed). Replaces the process-wide bucket.
        redisPool := apiutil.RedisPool()
        if redisPool != nil {
                pingCtx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
                if err := ratelimit.Ping(pingCtx, redisPool); err != nil {
                        logging.Logger.Error("REDIS_HOST is configured but unreachable at startup; rate-limited requests will fail closed until it recovers",
                                "error", err.Error())
                }
                cancel()
        } else {
                logging.Logger.Warn("REDIS_HOST is not configured; rate limiting will fail closed (503) on every limited request")
        }
        r.Use(ratelimit.Middleware(redisPool, ratelimit.DefaultOptions()))

        authzClient := authz.NewClient(cfg.AuthAPIURL, cfg.UsersAPIURL)
        catalogSvc := catalog.New(store)
        bill := billing.New(store, billing.NewStripeGateway(cfg.StripeSecretKey), catalogSvc, billing.URLs{
                CheckoutSuccess: cfg.CheckoutSuccessURL, CheckoutCancel: cfg.CheckoutCancelURL, PortalReturn: cfg.PortalReturnURL,
        })
        server.SetupHandlers(r, catalogSvc, bill, authzClient, cfg)

        _ = r.Run(util.GetEnv(apiconstants.BIND_ADDRESS, ":8000"))
}

                        Dsn:              sentryDsn,
                        Debug:            sentryDebug,

// setupProfiling starts continuous profiling only when the profiling-enabled
// feature flag evaluates to true, regardless of whether
// PYROSCOPE_SERVER_ADDRESS happens to be set - the flag is the on/off
// control, PYROSCOPE_SERVER_ADDRESS is only the destination. See the
// pyroscope-profiling-flag spec's three scenarios.
func setupProfiling(ff *featureflags.Client) func() {
        logging.Logger.Info("Setting up continuous profiling...")

        if !ff.BoolFlag(context.Background(), constants.ProfilingEnabledFlag, false) {
                logging.Logger.Info("profiling-enabled flag is off, continuous profiling disabled")
                return nil
        }

        serverAddress, found := os.LookupEnv(constants.PYROSCOPE_SERVER_ADDRESS)
        if !found {
                logging.Logger.Warn("profiling-enabled flag is on but PYROSCOPE_SERVER_ADDRESS not set, continuous profiling disabled")
                return nil
        }

        profiler, err := pyroscope.Start(pyroscope.Config{
                ApplicationName: constants.ServiceName,
                ServerAddress:   serverAddress,
                TenantID:        util.GetEnv(constants.PYROSCOPE_TENANT_ID, ""),
                Tags: map[string]string{
                        "env": util.GetEnv(apiconstants.ENV, "dev"),
                },
        })
        if err != nil {
                logging.Logger.Error("Error while trying to initialize continuous profiling", "error", err.Error())
                return nil
        }

        return func() {
                _ = profiler.Stop()
        }
}

package config

import (
        "errors"
        "net/url"
        "strings"

        "github.com/sweetrpg/entitlements-api/constants"
)

// Config holds validated settings.
type Config struct {
        AuthAPIURL  string
        UsersAPIURL string

        StripeSecretKey     string
        StripeWebhookSecret string
        CheckoutSuccessURL  string
        CheckoutCancelURL   string
        PortalReturnURL     string
}

// Load reads and validates settings through getenv (os.Getenv in production). Every problem is
// reported at once, naming the setting and never its value.
func Load(getenv func(string) string) (*Config, error) {
        var problems []string

        cfg := &Config{
                AuthAPIURL:          getenv(constants.AUTH_API_URL),
                UsersAPIURL:         getenv(constants.USERS_API_URL),
                StripeSecretKey:     getenv(constants.STRIPE_SECRET_KEY),
                StripeWebhookSecret: getenv(constants.STRIPE_WEBHOOK_SECRET),
                CheckoutSuccessURL:  getenv(constants.STRIPE_CHECKOUT_SUCCESS_URL),
                CheckoutCancelURL:   getenv(constants.STRIPE_CHECKOUT_CANCEL_URL),
                PortalReturnURL:     getenv(constants.STRIPE_PORTAL_RETURN_URL),
        }
        for _, name := range []string{constants.AUTH_API_URL, constants.USERS_API_URL, constants.STRIPE_SECRET_KEY, constants.STRIPE_WEBHOOK_SECRET} {
                if getenv(name) == "" {
                        problems = append(problems, name+" is required")
                }
        }

        if cfg.StripeSecretKey != "" && !hasPrefix(cfg.StripeSecretKey, "sk_", "rk_") {
                problems = append(problems, constants.STRIPE_SECRET_KEY+" must be a Stripe secret or restricted key")
        }
        if cfg.StripeWebhookSecret != "" && !hasPrefix(cfg.StripeWebhookSecret, "whsec_") {
                problems = append(problems, constants.STRIPE_WEBHOOK_SECRET+" must be a Stripe webhook signing secret")
        }
        for name, v := range map[string]string{
                constants.STRIPE_CHECKOUT_SUCCESS_URL: cfg.CheckoutSuccessURL,
                constants.STRIPE_CHECKOUT_CANCEL_URL:  cfg.CheckoutCancelURL,
                constants.STRIPE_PORTAL_RETURN_URL:    cfg.PortalReturnURL,
        } {
                if !absoluteURL(v) {
                        problems = append(problems, name+" must be an absolute http(s) URL")
                }
        }

        if getenv("DB_URI") == "" && getenv("DB_HOST") == "" {
                problems = append(problems, "DB_URI (or DB_HOST and DB_NAME) is required")
        }

        if len(problems) > 0 {
                return nil, errors.New("invalid configuration: " + strings.Join(problems, "; "))
        }
        return cfg, nil
}

func hasPrefix(v string, prefixes ...string) bool {
        for _, p := range prefixes {
                if strings.HasPrefix(v, p) {
                        return true
                }
        }
        return false
}

func absoluteURL(v string) bool {
        u, err := url.Parse(v)
        return err == nil && (u.Scheme == "https" || u.Scheme == "http") && u.Host != ""
}

package data

import (
        "context"
        "slices"
        "strings"
        "sync"
        "time"

        "github.com/sweetrpg/entitlements-api/models"
        mcore "github.com/sweetrpg/model-core.go/models"
)

// MemoryStore is an in-process Store used by tests so they need no database. It mirrors the
// Mongo store's semantics: soft deletes, live-only uniqueness, and the same grant activity rule.
type MemoryStore struct {
        mu      sync.Mutex
        keys    []*models.Key
        tiers   []*models.Tier
        prods   []*models.Product
        grants  []models.Grant
        audit   []models.AuditEntry
        custs   map[string]string
        events  map[string]bool
        defTier string
        now     func() time.Time
}

// NewMemoryStore returns an empty store using the wall clock.
func NewMemoryStore() *MemoryStore { return &MemoryStore{now: time.Now} }

// AddGrant seeds a grant for tests.
func (m *MemoryStore) AddGrant(g models.Grant) {
        m.mu.Lock()
        defer m.mu.Unlock()
        m.grants = append(m.grants, g)
}

func live(a mcore.Auditable) bool { return a.DeletedAt == nil }

func (m *MemoryStore) InsertKey(_ context.Context, k *models.Key, actor string) error {
        m.mu.Lock()
        defer m.mu.Unlock()
        if m.findKey(k.Key) != nil {
                return ErrDuplicate
        }
        mcore.StampCreate(&k.Auditable, actor, m.now().UTC())
        c := *k
        m.keys = append(m.keys, &c)
        return nil
}

func (m *MemoryStore) findKey(name string) *models.Key {
        for _, k := range m.keys {
                if k.Key == name && live(k.Auditable) {
                        return k
                }
        }
        return nil
}

func (m *MemoryStore) GetKeys(_ context.Context, names []string) ([]models.Key, error) {
        m.mu.Lock()
        defer m.mu.Unlock()
        var out []models.Key
        for _, k := range m.keys {
                if live(k.Auditable) && slices.Contains(names, k.Key) {
                        out = append(out, *k)
                }
        }
        return out, nil
}

func (m *MemoryStore) ListKeys(_ context.Context, prefix string, limit, skip int64) ([]models.Key, error) {
        m.mu.Lock()
        defer m.mu.Unlock()
        var out []models.Key
        for _, k := range m.keys {
                if live(k.Auditable) && strings.HasPrefix(k.Key, prefix) {
                        out = append(out, *k)
                }
        }
        slices.SortFunc(out, func(a, b models.Key) int { return strings.Compare(a.Key, b.Key) })
        return window(out, limit, skip), nil
}

func (m *MemoryStore) UpdateKey(_ context.Context, name, description string, ttl int, max *int64, actor string) (*models.Key, error) {
        m.mu.Lock()
        defer m.mu.Unlock()
        k := m.findKey(name)
        if k == nil {
                return nil, ErrNotFound

// window applies skip then limit; limit <= 0 means no limit.
func window[T any](s []T, limit, skip int64) []T {
        if skip >= int64(len(s)) {
                return nil
        }
        s = s[skip:]
        if limit > 0 && limit < int64(len(s)) {
                s = s[:limit]
        }
        return s
}

// MongoStore implements Store on MongoDB. It is a thin translation layer: the rules live in the
// catalog package and are tested against MemoryStore; this file is exercised against a live
// database only.
type MongoStore struct {
        db *mongo.Database
}

// NewMongoStore wraps db.
func NewMongoStore(db *mongo.Database) *MongoStore { return &MongoStore{db: db} }

// notDeleted is merged into every catalog read so a new read path inherits the tombstone filter.
var notDeleted = bson.E{Key: "deleted_at", Value: nil}

// indexSpecs is every index the service relies on, by collection. It is a pure function so a test
// can check each definition against what MongoDB accepts without a database.
func indexSpecs() map[string][]mongo.IndexModel {
        // MongoDB partial indexes cannot express "deleted_at is absent" ($exists: false is rejected),
        // so live documents carry live: true, set on insert and removed on soft delete.
        liveOnly := options.Index().SetUnique(true).
                SetPartialFilterExpression(bson.D{{Key: "live", Value: true}})
        return map[string][]mongo.IndexModel{
                constants.KeysCollection:  {{Keys: bson.D{{Key: "key", Value: 1}}, Options: liveOnly}},
                constants.TiersCollection: {{Keys: bson.D{{Key: "slug", Value: 1}}, Options: liveOnly}},
                constants.ProductsCollection: {
                        {Keys: bson.D{{Key: "slug", Value: 1}}, Options: liveOnly},
                        {Keys: bson.D{{Key: "stripe_price_id", Value: 1}}},
                },
                constants.GrantsCollection: {
                        {Keys: bson.D{{Key: "user_id", Value: 1}, {Key: "status", Value: 1}}},
                        {Keys: bson.D{{Key: "source", Value: 1}, {Key: "external_id", Value: 1}, {Key: "key", Value: 1}}, Options: options.Index().SetUnique(true).
                                SetPartialFilterExpression(bson.D{{Key: "external_id", Value: bson.D{{Key: "$exists", Value: true}}}})},
                        {Keys: bson.D{{Key: "refs", Value: 1}}},
                },
                constants.StripeCustomersCollection: {{Keys: bson.D{{Key: "user_id", Value: 1}}, Options: options.Index().SetUnique(true)}},
                constants.AuditCollection:           {{Keys: bson.D{{Key: "at", Value: -1}}}},
        }
}

// EnsureIndexes creates the unique-among-live indexes and the lookup indexes. Safe to call on
// every startup.
func (s *MongoStore) EnsureIndexes(ctx context.Context) error {
        // The first grants index was unique on (source, external_id) alone, which would block a
        // bundle's several grants sharing one external ID. Drop it; a missing index is not an error.
        _, _ = s.db.Collection(constants.GrantsCollection).Indexes().DropOne(ctx, "source_1_external_id_1")
        if err := s.backfillLive(ctx); err != nil {
                return err
        }
        for coll, ms := range indexSpecs() {
                if _, err := s.db.Collection(coll).Indexes().CreateMany(ctx, ms); err != nil {
                        return fmt.Errorf("indexes on %s: %w", coll, err)
                }
        }
        return nil
}

// liveCollections are the soft-deleted catalog collections whose uniqueness is among live rows.
var liveCollections = []string{constants.KeysCollection, constants.TiersCollection, constants.ProductsCollection}

// backfillLive marks documents written before the live flag existed. It is idempotent and only
// touches live documents that lack the flag, so it is a no-op after the first startup.
func (s *MongoStore) backfillLive(ctx context.Context) error {
        for _, coll := range liveCollections {
                f := bson.D{{Key: "live", Value: bson.D{{Key: "$exists", Value: false}}}, notDeleted}
                if _, err := s.db.Collection(coll).UpdateMany(ctx, f, bson.D{{Key: "$set", Value: bson.D{{Key: "live", Value: true}}}}); err != nil {
                        return fmt.Errorf("backfill live on %s: %w", coll, err)
                }
        }
        return nil
}

func (s *MongoStore) insert(ctx context.Context, coll string, doc any) error {
        _, err := s.db.Collection(coll).InsertOne(ctx, doc)
        if mongo.IsDuplicateKeyError(err) {
                return ErrDuplicate
        }
        return err
}

func find[T any](ctx context.Context, c *mongo.Collection, filter bson.D, opts *options.FindOptions) ([]T, error) {
        cur, err := c.Find(ctx, filter, opts)
        if err != nil {
                return nil, err
        }
        var out []T
        if err := cur.All(ctx, &out); err != nil {
                return nil, err
        }
        return out, nil
}

func page(sortKey string, dir int, limit, skip int64) *options.FindOptions {
        o := options.Find().SetSort(bson.D{{Key: sortKey, Value: dir}}).SetSkip(skip)
        if limit > 0 {
                o.SetLimit(limit)
        }
        return o
}

func (s *MongoStore) softDelete(ctx context.Context, coll string, filter bson.D, actor string) error {
        set := bson.D{{Key: "deleted_at", Value: time.Now().UTC()}, {Key: "deleted_by", Value: actor}}
        update := bson.D{{Key: "$set", Value: set}, {Key: "$unset", Value: bson.D{{Key: "live", Value: ""}}}}
        res, err := s.db.Collection(coll).UpdateOne(ctx, filter, update)
        if err != nil {
                return err
        }
        if res.MatchedCount == 0 {
                return ErrNotFound
        }
        return nil
}

// ValidateKeyName checks the namespaced key format, e.g. initiative.theme_slots.
func ValidateKeyName(name string) error {
        if len(name) > 128 || !keyPattern.MatchString(name) {
                return fmt.Errorf("%w: key must be namespaced lowercase, like service.feature_name", ErrInvalid)
        }
        return nil
}

// ValidateTierSlug checks a tier identifier.
func ValidateTierSlug(slug string) error {
        if !tierPattern.MatchString(slug) {
                return fmt.Errorf("%w: tier id must be lowercase letters, digits, - or _ (max 64)", ErrInvalid)
        }
        return nil
}

// ValidateProductSlug checks a product identifier; products share the tier id format.
func ValidateProductSlug(slug string) error { return ValidateTierSlug(slug) }

// ValidateMax checks a key's maximum: only limit keys carry one, and it must be a positive
// integer that fits a limit value.
func ValidateMax(t KeyType, max *int64) error {
        if max == nil {
                return nil
        }
        if t != TypeLimit {
                return fmt.Errorf("%w: only limit keys can have a maximum", ErrInvalid)
        }
        if *max < 1 || *max > math.MaxInt32 {
                return fmt.Errorf("%w: maximum must be between 1 and %d", ErrInvalid, math.MaxInt32)
        }
        return nil
}

// ParseValue converts a JSON value to its stored form: flags take true/false, limits take a
// non-negative integer.
func ParseValue(t KeyType, raw any) (int64, error) {
        switch t {
        case TypeFlag:
                b, ok := raw.(bool)
                if !ok {
                        return 0, fmt.Errorf("%w: flag values must be true or false", ErrInvalid)
                }
                if b {
                        return 1, nil
                }
                return 0, nil
        case TypeLimit:
                f, ok := raw.(float64)
                if !ok || f < 0 || f != math.Trunc(f) || f > math.MaxInt32 {
                        return 0, fmt.Errorf("%w: limit values must be non-negative integers", ErrInvalid)
                }
                return int64(f), nil
        }
        return 0, fmt.Errorf("%w: unknown key type %q", ErrInvalid, t)
}

// FormatValue is the inverse of ParseValue.
func FormatValue(t KeyType, v int64) any {
        if t == TypeFlag {
                return v > 0
        }
        return v
}

// paging reads limit/skip query parameters, clamping limit to a sane page.
func paging(c *gin.Context) (limit, skip int64) {
        limit, _ = strconv.ParseInt(c.DefaultQuery("limit", "50"), 10, 64)

// checkoutHandler starts a purchase or switches a plan.

//        @Summary                Start a checkout or switch a plan
//        @Description        Returns a Stripe-hosted checkout URL, or switches the caller's existing subscription in the product's plan group. A switch needs confirm=true: without it the response is 409 confirmation_required carrying a quote to show the user; an upgrade is charged immediately and a failed payment is 402 payment_failed with nothing changed.
//        @Tags                        billing
//        @Accept                        json
//        @Produce                json
//        @Param                        body        body                checkoutBody        true        "tier_id or product_id (+ quantity, confirm, proration_date)"
//        @Success                200                {object}        SessionVO
//        @Failure                400                {object}        vo.ErrorVO
//        @Failure                402                {object}        vo.ErrorVO
//        @Failure                409                {object}        ConfirmationRequiredVO
//        @Failure                502                {object}        vo.ErrorVO

                })
                if err != nil {
                        writeError(c, err)
                        return
                }
                c.JSON(http.StatusOK, SessionVO{URL: res.URL, Switched: res.Switched})
        }
}

// setupWebhookHandlers registers the public Stripe webhook. It sits outside the user-token
// groups on purpose: Stripe cannot present a user token, so the signature over the raw body is
// the only authentication, and nothing is read or changed before it verifies.
func setupWebhookHandlers(g *gin.Engine, svc *billing.Service, signingSecret string) {
        g.POST("/webhooks/stripe", func(c *gin.Context) {
                body, err := io.ReadAll(http.MaxBytesReader(c.Writer, c.Request.Body, constants.MaxWebhookBodyBytes))
                if err != nil {
                        badRequest(c, "unreadable body")
                        return
                }
                // Event objects are decoded by the handlers' own minimal structs, so the SDK's API-version
                // compatibility check would only reject events for fields we never read.
                ev, err := webhook.ConstructEventWithOptions(body, c.GetHeader("Stripe-Signature"), signingSecret,
                        webhook.ConstructEventOptions{IgnoreAPIVersionMismatch: true})
                if err != nil {
                        c.JSON(http.StatusBadRequest, apiv.ErrorVO{Error: "invalid_signature", Message: "signature verification failed"})
                        return
                }
                err = svc.HandleEvent(c.Request.Context(), billing.Event{ID: ev.ID, Type: string(ev.Type), Object: ev.Data.Raw})
                if err != nil {
                        logging.Logger.Error("stripe event failed", "event", ev.ID, "type", string(ev.Type), "error", err.Error())
                        c.JSON(http.StatusInternalServerError, apiv.ErrorVO{Error: "internal_error", Message: "event not processed"})
                        return
                }
                c.Status(http.StatusOK)
        })
}

// tokenEmail reads the email claim from a token auth-api has already verified, to prefill
// Checkout. It does not verify anything itself and returns "" when the claim is absent.
func tokenEmail(token string) string {
        parts := strings.Split(token, ".")
        if len(parts) != 3 {
                return ""
        }
        payload, err := base64.RawURLEncoding.DecodeString(parts[1])
        if err != nil {
                return ""
        }
        var claims struct {
                Email string `json:"email"`
        }
        if json.Unmarshal(payload, &claims) != nil {
                return ""
        }
        return claims.Email
}

// CheckVO is one key's effective value. Consumers may cache it for TTLSeconds.
type CheckVO struct {
        Key        string         `json:"key"`
        Type       models.KeyType `json:"type"`
        Value      any            `json:"value"`
        TTLSeconds int            `json:"ttl_seconds"`
}

// BatchCheckVO is the response to a batch check.
type BatchCheckVO struct {
        Results []CheckVO `json:"results"`
}

func toCheckVO(r catalog.Result) CheckVO {
        return CheckVO{Key: r.Key, Type: r.Type, Value: models.FormatValue(r.Type, r.Value), TTLSeconds: r.TTLSeconds}
}

// anyRole lists every platform role: a check needs a verified user, not a particular role.
var anyRole = []string{authz.RoleUser, authz.RoleSubmitter, authz.RoleEditor, authz.RoleModerator, authz.RoleApprover, authz.RoleAdmin}

// setupCheckHandlers registers the check routes. The caller passes the end user's own bearer
// token; auth-api verifies it against Auth0 and users-api resolves the canonical user ID, so the
// answer is always about the verified caller and no user ID is accepted from the request.
func setupCheckHandlers(g *gin.Engine, svc *catalog.Service, client *authz.Client) {
        s := g.Group("/check", authz.RequireAnyRole(client, constants.ServiceName, anyRole...))

        s.GET("", checkOne(svc))
        s.GET("/batch", checkBatch(svc))

// checkOne returns the caller's effective value for one key.
//
//        @Summary                Check one key
//        @Description        Effective value for one key for the verified caller. Cacheable for ttl_seconds.
//        @Tags                        check
//        @Produce                json
//        @Param                        key        query                string        true        "Entitlement key, e.g. initiative.theme_slots"
//        @Success                200        {object}        CheckVO
//        @Failure                400        {object}        vo.ErrorVO
//        @Failure                401        {object}        vo.ErrorVO
//        @Failure                404        {object}        vo.ErrorVO
//        @Router                        /check [get]
func checkOne(svc *catalog.Service) gin.HandlerFunc {
        return func(c *gin.Context) {
                key := c.Query("key")
                if key == "" {
                        badRequest(c, "key is required")
                        return
                }
                results, err := svc.Check(c.Request.Context(), authz.Viewer(c), []string{key})
                if err != nil {
                        writeError(c, err)
                        return
                }
                vo := toCheckVO(results[0])
                c.Header("Cache-Control", cacheControl(vo.TTLSeconds))
                c.JSON(http.StatusOK, vo)
        }
}

// checkBatch returns effective values for several keys.
//
//        @Summary                Check several keys
//        @Description        Exactly one of repeated key params or a prefix. An unknown key fails the whole batch with 404.
//        @Tags                        check
//        @Produce                json
//        @Param                        key                query                []string        false        "Entitlement key; repeat for several (max 200)"        collectionFormat(multi)
//        @Param                        prefix        query                string                false        "Key prefix, e.g. initiative. (max 128 characters)"
//        @Success                200                {object}        BatchCheckVO
//        @Failure                400                {object}        vo.ErrorVO
//        @Failure                401                {object}        vo.ErrorVO
//        @Failure                404                {object}        vo.ErrorVO
//        @Router                        /check/batch [get]
func checkBatch(svc *catalog.Service) gin.HandlerFunc {
        return func(c *gin.Context) {
                keys, prefix := c.QueryArray("key"), c.Query("prefix")
                if code, msg := validateBatchSelector(keys, prefix); code != "" {
                        c.JSON(http.StatusBadRequest, apiv.ErrorVO{Error: code, Message: msg})
                        return
                }
                var results []catalog.Result
                var err error
                if prefix != "" {
                        results, err = svc.CheckPrefix(c.Request.Context(), authz.Viewer(c), prefix)
                } else {
                        results, err = svc.Check(c.Request.Context(), authz.Viewer(c), keys)
                }
                if err != nil {
                        writeError(c, err)
                        return
                }
                out := BatchCheckVO{Results: make([]CheckVO, len(results))}
                for i, r := range results {
                        out.Results[i] = toCheckVO(r)
                }
                c.JSON(http.StatusOK, out)
        }
}

// validateBatchSelector requires exactly one of repeated key params or a prefix, within the
// caps in constants. It returns an error code and message, or "" when the selector is valid.
func validateBatchSelector(keys []string, prefix string) (code, msg string) {
        switch {
        case (len(keys) == 0) == (prefix == ""):
                return "invalid_selector", "provide either key (repeatable) or prefix, not both and not neither"
        case len(keys) > constants.MaxBatchKeys:
                return "too_many_keys", fmt.Sprintf("at most %d keys per request", constants.MaxBatchKeys)
        case len(prefix) > constants.MaxPrefixLength:
                return "prefix_too_long", fmt.Sprintf("prefix may be at most %d characters", constants.MaxPrefixLength)
        }
        return "", ""
}

func cacheControl(ttl int) string {
        if ttl <= 0 {
                return "no-store"
        }
        return fmt.Sprintf("private, max-age=%d", ttl)
}

        UnitsRemaining int64  `json:"units_remaining"`

// writeError maps a service error to a response. Unexpected errors are logged and reported
// without detail.
func writeError(c *gin.Context, err error) {
        var unknown catalog.UnknownKeysError
        var exceeded catalog.MaxExceededError
        var provider *billing.ProviderError
        var paymentFailed *billing.PaymentFailedError
        var confirm *billing.ConfirmationRequiredError
        var noIncrease catalog.NoIncreaseError
        switch {
        case errors.As(err, &confirm):
                c.JSON(http.StatusConflict, ConfirmationRequiredVO{
                        Error: "confirmation_required", Message: "confirm the plan change to go ahead", Quote: confirm.Quote,
                })
        case errors.As(err, &paymentFailed):
                c.JSON(http.StatusPaymentRequired, apiv.ErrorVO{Error: "payment_failed", Message: paymentFailed.PublicMessage()})
        case errors.As(err, &provider):
                c.JSON(http.StatusBadGateway, apiv.ErrorVO{Error: "payment_provider_error", Message: provider.PublicMessage()})
        case errors.As(err, &noIncrease):
                c.JSON(http.StatusConflict, apiv.ErrorVO{Error: "no_increase", Message: err.Error()})
        case errors.As(err, &exceeded):
                c.JSON(http.StatusConflict, MaxExceededVO{Error: "exceeds_maximum", Message: err.Error(), Key: exceeded.Key, UnitsRemaining: exceeded.Remaining})
        case errors.Is(err, models.ErrInvalid):
                c.JSON(http.StatusBadRequest, apiv.ErrorVO{Error: "invalid_request", Message: err.Error()})
        case errors.As(err, &unknown):
                c.JSON(http.StatusNotFound, apiv.ErrorVO{Error: "unknown_key", Message: err.Error()})
        case errors.Is(err, catalog.ErrNotFound):
                c.JSON(http.StatusNotFound, apiv.ErrorVO{Error: "not_found", Message: "not found"})
        case errors.Is(err, catalog.ErrDuplicate):
                c.JSON(http.StatusConflict, apiv.ErrorVO{Error: "duplicate", Message: "already exists"})
        case errors.Is(err, catalog.ErrInUse):
                c.JSON(http.StatusConflict, apiv.ErrorVO{Error: "in_use", Message: err.Error()})
        default:
                logging.Logger.Error("request failed", "error", err.Error())
                c.JSON(http.StatusInternalServerError, apiv.ErrorVO{Error: "internal_error", Message: "internal error"})
        }
}

func badRequest(c *gin.Context, msg string) {
        c.JSON(http.StatusBadRequest, apiv.ErrorVO{Error: "invalid_request", Message: msg})
}

func setupStatusHandlers(g *gin.Engine) {
        logging.Logger.Info("Setting up status endpoint handlers...")

        {
                a := g.Group("/status")
                a.GET("/health", healthHandler)
        }

        g.GET("/status/ping", pingHandler)
}

// Health check.
//
//         Do a health check of the application and its dependencies and return the results
//                @Summary                Health check
//                @Description        Health check
//                @Tags                        status
//                @Produce                json
//                @Success                200                {object}        vo.HealthResponseVO
//                @Router                        /status/health [get]
func healthHandler(c *gin.Context) {
        resp := apicores.HealthHandler(c.Request.Context())
        status := http.StatusOK
        if resp.Errors > 0 {
                status = http.StatusServiceUnavailable
        }
        c.JSON(status, resp)
}

// Ping the application.
//
//         A shallow health check to see if basic responses are working. Returns the date and hostname of the server handling the request.
//                @Summary                Ping
//                @Description        Ping
//                @Tags                        status
//                @Produce                json
//                @Success                200                {object}        vo.PingResponseVO
//                @Router                        /status/ping [get]
func pingHandler(c *gin.Context) {
        c.JSON(http.StatusOK, apicores.PingHandler())
}
